\section[HouseholdStages]{HouseholdStages}
% =============================================================

\begin{frame}{HouseholdStages}
  \begin{itemize}
    \item An implementation of Stages for building household blocks of heterogeneous-agent macro models.
    \item Auclert--Bardóczy--Rognlie--Straub \textbf{SSJ toolkit} (Python) also has stage implementations.
    \item Econ-ARK also has a form of Stages.
  \end{itemize}
\end{frame}

\begin{frame}{What the package gives you}
  \begin{itemize}
    \item \textbf{Stage implementations.} \texttt{MarkovStage}, \texttt{WealthChangeStage}, \texttt{ConsumptionSavingsStage}, \texttt{LogitChoiceStage}, \texttt{MigrationStage}, \texttt{BorrowingConstraintStage}, \texttt{AssetPriceChangeStage} $\dots$
    \item \textbf{Composition.} \texttt{$\circ$} chains Stages; \texttt{lift\_moments} attaches integrals.
    \item \textbf{Inner-solve helpers.} Steady state and transition V, $\Lambda$ given env (prices and params).
    \item \textbf{Functorial lifts.} \texttt{lift\_jacobian} (auto-diff), \texttt{lift\_gpu}.
    \item \textbf{SSJ utilities.} Sequence-space Jacobian (covered in L6+).
  \end{itemize}
\end{frame}

\begin{frame}{What the package does \emph{not} do}
  The library handles \textbf{the household block at a given env}. It does \emph{not}:
  \begin{itemize}
    \item Decide your aggregate state.
    \item Compute prices from $\bar K$ (firm block).
    \item Run outer loops for calibration or market clearing.
  \end{itemize}
\end{frame}

\begin{frame}{Why this split}
  \begin{itemize}
    \item Household block: \textbf{generic components} across HA models.
    \item Equilibrium, calibration, transition: \textbf{model-specific}.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Steady-state inner-solve helpers}
  \begin{lstlisting}
  # V backward to a fixed point at fixed env.
  solve_vfi_steady_state_given_env!(hh, env)

  # Λ forward to stationarity using the policy populated by the backward sweep.
  solve_lambda_steady_state_given_env!(hh)

  # Bundle of the two: V then Λ, plus attached moments.
  solve_steady_state_given_env!(hh, env)
  \end{lstlisting}
  All three take \texttt{env} \textbf{as given}. They iterate the household block but do not update prices or parameters.
\end{frame}

\section{Aiyagari Steady State in Stages}
% =============================================================

\begin{frame}{Same Model as Before}
  \begin{itemize}
    \item \textbf{Stage Chain} (time order): \[ \text{z\_shock} \circ \text{income} \circ \text{savings}. \]
    \item \textbf{Firm.} Cobb-Douglas, fixed labor $L$.
    \item \textbf{Equilibrium.} $R = R^{\mathrm{implied}}(K^{\mathrm{supplied}}(R))$.
    \item \textbf{Outer loop.} Damped \textbf{tatonnement} on $R$.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{State Layout}
  \begin{lstlisting}
  layout = StateLayout(
      StateAxis(:wealth, continuous_grid(0, 100; length=400, spacing=:log)),
      StateAxis(:z,      [0.6, 1.0, 1.4]),
  )
  \end{lstlisting}
  \begin{itemize}
    \item \textbf{wealth} (continuous, 400-pt log grid).
    \item \textbf{z} (productivity; discrete, 3 levels).
    \item Closures read \texttt{cell.wealth}, \texttt{cell.z}.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Stage 1: $z$ Shock}
  \begin{lstlisting}
  z_shock = MarkovStage(layout; axis=:z, transition=p.Π_z)
  \end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Stage 2: Income}
  \begin{lstlisting}
  income = WealthChangeStage(layout; wealth_post=(cell; env) -> env.R * cell.wealth + env.w * cell.z)
  \end{lstlisting}
  \begin{itemize}
    \item Previous example snapped-to-grid. The library function reinterpolates continuously.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Stage 3: Consumption Savings}
  \begin{lstlisting}
  savings = ConsumptionSavingsStage(layout; β=p.β, utility=(cell, c; env) -> u_crra(c, Val(p.σ)), monotone_search=:divide_conquer)
  \end{lstlisting}
  \begin{itemize}
    \item \texttt{:divide\_conquer}: faster $O(N \log N)$ algorithm for when policy is monotone in $b$.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Compose}
  \begin{lstlisting}
  hh = z_shock ∘ income ∘ savings
  \end{lstlisting}
  \begin{itemize}
    \item \texttt{$\circ$} composes Stages in time order.
    \item The household block is \emph{itself a Stage}: own \texttt{backward!} and \texttt{forward!}.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Define Moments}
  \begin{lstlisting}
  hh = define_moments!(hh; K_supplied=at_end(integrand=:wealth, reduce=sum))
  \end{lstlisting}
  \begin{itemize}
    \item \texttt{define\_moments!}: Define integrals on $\Lambda$. Here $K^{\mathrm{supplied}} = \int b\,\mathrm{d}\Lambda^\text{end}$.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Solve Steady State Given $R$}
  \begin{lstlisting}
  (;K, w) = aiyagari_supply_side(R, p)
  env = make_env(hh; R, w)
  res = solve_steady_state_given_env!(hh, env)
  K_supplied = res.moments.K_supplied
  \end{lstlisting}
  \begin{itemize}
    \item \textbf{Line 1.} From $R$, get firm-side $K$ and wage $w$.
    \item \textbf{Line 2.} Build env: $(R, w)$.
    \item \textbf{Line 3.} Find steady state $V$ and $\Lambda$.
    \item \textbf{Line 4.} Read off $K^{\mathrm{supplied}}$.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Outer Tatonnement Loop}
  \begin{lstlisting}
  R = R_init
  while err >= rtol
      env = make_env(hh; R, w=aiyagari_supply_side(R, p).w)
      (;V, Λ, moments) = solve_steady_state_given_env!(hh, env)
      R_implied = aiyagari_prices(moments.K_supplied, p).R
      err = abs(R_implied - R)
      R += update_speed*(R_implied - R)           # damped tatonnement
  end
  \end{lstlisting}
  Each pass: $R \to (K, w) \to V,\Lambda \to K^{\mathrm{supplied}} \to R^{\mathrm{implied}}$, then update $R$.

  Stop when $|R^{\mathrm{implied}} - R| < $\texttt{rtol}.

  \vspace{0.4em}
  \footnotesize Tatonnement on the \emph{price} is the general approach: the household block is a price-taker, and the outer loop searches for the price that clears the market. Tatonnement on $\bar K$ happens to work in Aiyagari because asset- and goods-market clearing coincide, but it does not generalize.
\end{frame}

% =============================================================
\section[Comparative statics]{Comparative statics}
% =============================================================

\begin{frame}{Comparative Statics}
  Change one parameter and re-solve steady state.

  Can think of this as the \textbf{long-run} response to a shock.

  In the example calibration:
  \begin{itemize}
    \item $A = 1$ \,$\to$\, $\bar K_{\mathrm{pre}} = [\text{verify}]$, $\bar R_{\mathrm{pre}} = [\text{verify}]$.
    \item $A = 1.05$ \,$\to$\, $\bar K_{\mathrm{post}} = [\text{verify}]$, $\bar R_{\mathrm{post}} = [\text{verify}]$.
  \end{itemize}
\end{frame}

\begin{frame}{Comparative Statics}
  Comparative statics tells you \textbf{long-run} response to shocks.

  Doesn't tell you:
  \begin{itemize}
    \item Speed of transition
    \item Path of transition
    \item Welfare impacts of shock on those alive at the time.
  \end{itemize}

  For these, we need transition dynamics.
\end{frame}

% =============================================================
\section[Transition Dynamics]{Transition Dynamics}
% =============================================================

\begin{frame}{Transition Dynamics}
  \begin{itemize}
    \item \textbf{Steady state.} A single $(K, V, \Lambda)$. Time-invariant.
    \item \textbf{Transition.} A sequence $\{(K_t, V_t, \Lambda_t)\}_{t = 1,\dots,T}$. The economy is \emph{moving}.
  \end{itemize}
\end{frame}

\begin{frame}{The MIT Shock}
  A \textbf{permanent, unexpected} shock at $t = 1$.
  \begin{itemize}
    \item $t = 0$: At steady state.
    \item $t = 1$: Parameter permanently changes. Agents are \emph{surprised}.
    \item $t \geq 1$: \emph{Perfect foresight} of the entire future path of exogenous and endogenous variables.
  \end{itemize}
\end{frame}

\begin{frame}{Example: TFP Shock}
  \emph{Permanent} positive TFP shock at $t = 1$:
  \[
    A_t = 1.05 \quad \text{for all } t \geq 1, \quad A_0 = 1.
  \]
  \begin{itemize}
    \item The \emph{only} exogenous change.
    \item $\{R_t, w_t, K_t, V_t, \Lambda_t\}$ change in \textbf{response}.
    \item Set $T = 100$ and set $V_{T+1} = V^\text{new\_steady\_state}$ to close the model.
  \end{itemize}
\end{frame}

% =============================================================
\section[Solving MIT transition]{Solving MIT transition}
% =============================================================

\begin{frame}{The problem statement}
  \begin{itemize}
    \item \textbf{Given:} $\{A_t\}_{t = 1}^T$.
    \item \textbf{Find:} $\{R_t, V_t, \Lambda_t\}$ such that
  \end{itemize}
  \begin{itemize}
    \item $V_{T+1} = V_{\mathrm{ss}^{\mathrm{post}}}$ (terminal).
    \item $\Lambda_1 = \Lambda_{\mathrm{ss}^{\mathrm{pre}}}$ (initial).
    \item $V_t = \text{backward}(V_{t+1}, \mathrm{env}_t)$.
    \item $\Lambda_{t+1} = \text{forward}(\Lambda_t, V_{t+1}, \mathrm{env}_t)$.
    \item Markets clear: $R_t = R^{\mathrm{implied}}(K_t^{\mathrm{supplied}})$.
  \end{itemize}
\end{frame}

\begin{frame}{Algorithm}

\textbf{1. Guess path $\{R_t\}$.}

\textbf{2. Iterate $V$ backward.}
\[
  V_{T+1} \leftarrow V_{\mathrm{ss}^{\mathrm{post}}}, \quad V_t = \text{backward}(V_{t+1}, \mathrm{env}_t).
\]

\textbf{3. Simulate $\Lambda$ forward.}
\[
  \Lambda_1 \leftarrow \Lambda_{\mathrm{ss}^{\mathrm{pre}}}, \quad \Lambda_{t+1} = \text{forward}(\Lambda_t, V_{t+1}, \mathrm{env}_t).
\]

\textbf{4. Update $R_{\mathrm{path}}$.}
\[
  R_t \leftarrow (1 - s)\,R_t + s\,R_t^{\mathrm{implied}}, \quad R_t^{\mathrm{implied}} = R^{\mathrm{implied}}(K_t^{\mathrm{supplied}}).
\]

\end{frame}

\begin{frame}{Note: Rootfinding}
  Ultimately, the problem is to find the root of the excess demand function in sequence space:
  \[
    \text{excess\_demand}(\{R_t\}) = \vec{0}.
  \]
  For demonstration, we are using the simple and robust method of tatonnement.

  Gradient-based methods using the sequence-space Jacobian can be much faster.
\end{frame}

\begin{frame}[fragile]{The algorithm}
\begin{lstlisting}[basicstyle=\ttfamily\scriptsize]
hh_path = [aiyagari_household(p_new) for _ in 1:T]   # one chain per period
R_path = collect(range(pre.R, post.R; length=T))     # initial guess
V_path[T+1] .= post.V; Λ_path[1] .= pre.Λ            # boundary conditions
while err >= rtol
    env_path = [make_env(hh_path[t]; R=R_path[t], w=aiyagari_supply_side(R_path[t], p_new).w) for t in 1:T]
    for t in T:-1:1                                  # backward sweep
        copyto!(V_path[t], backward!(hh_path[t], V_path[t+1], env_path[t]))
    end
    for t in 1:T                                     # forward sweep
        copyto!(Λ_path[t+1], forward!(hh_path[t], Λ_path[t]))
        K_S = compute_moments(hh_path[t], Λ_path[t+1], env_path[t]).K_supplied
        R_implied_path[t] = aiyagari_prices(K_S, p_new).R
    end
    err = maximum(abs.(R_implied_path .- R_path))
    R_path .+= update_speed .* (R_implied_path .- R_path)                  # damped tatonnement
end
\end{lstlisting}
\end{frame}

\section[Implementation]{Implementation}
% =============================================================

\begin{frame}{Initial Steady State}

\textbf{We need both:}

\begin{itemize}
  \item $V_{\mathrm{ss}^{\mathrm{post}}}$: terminal for backward sweep.
  \item $\Lambda_{\mathrm{ss}^{\mathrm{pre}}}$: initial for forward sweep.
\end{itemize}

\vspace{0.4em}

\begin{itemize}
  \item Section 2's tatonnement gives both (run once at each end).
  \item \textbf{Setup:} $V_{\mathrm{ss}^{\mathrm{post}}} \to V_\mathrm{path}[T+1]$; $\Lambda_{\mathrm{ss}^{\mathrm{pre}}} \to \Lambda_\mathrm{path}[1]$; initial $\{R_t\}$ a straight line from $R^{\mathrm{pre}}_{\mathrm{ss}}$ to $R^{\mathrm{post}}_{\mathrm{ss}}$.
\end{itemize}

\end{frame}

\begin{frame}[fragile]{Backward sweep}
\begin{lstlisting}
for t in T:-1:1
    env_t = make_env(hh_path[t]; R=R_path[t], w=aiyagari_supply_side(R_path[t], p_new).w)

end
\end{lstlisting}

\end{frame}

for t in 1:T

\begin{frame}{IRF: capital}

\textbf{Shape.}

\begin{itemize}
  \item $\Delta A$ raises $R$.
  \item Higher $R$ $\to$ more savings.
  \item More saving $\to$ higher $K$ $\to$ lower $R$. Self-correcting.
\end{itemize}

\vspace{0.4em}

\textbf{Lag.} Wealth distribution has inertia. Peak in $K$ lags peak in $A$ by $\sim 5$ periods.

\end{frame}

\begin{frame}{IRF: $R_t$ and $w_t$}

\textbf{$R_t$.}

\begin{itemize}
  \item Jumps up on impact ($A$ rises before $K$ moves).
  \item Decays back toward $R^{\mathrm{post}}_{\mathrm{ss}} = 1/\beta$ as $K$ accumulates.
  \item In Aiyagari $R^{\mathrm{pre}}_{\mathrm{ss}} = R^{\mathrm{post}}_{\mathrm{ss}}$; $K$ absorbs the permanent shift.
\end{itemize}

\textbf{$w_t$.}

\begin{itemize}
  \item Same pattern, opposite sign on impact.
  \item Rises with $K$ to a permanently higher $w^{\mathrm{post}}_{\mathrm{ss}}$.
\end{itemize}

\end{frame}

\begin{frame}{Welfare}

\begin{itemize}
  \item Wealthy agents: transient capital gain on $R$.
  \item Wage-dependent agents: pick up the $w$ side.
\end{itemize}

\vspace{0.6em}

For Aiyagari we get a \emph{distribution} of welfare effects, period by period. Disentangling the two channels is one thing MIT shocks let you do.

\end{frame}

\begin{frame}{Pick $T$ large enough}

\begin{itemize}
  \item $T$ = horizon by which $\bar K$ and $V$ have converged to the post-shock steady state, so $V_{T+1} \approx V^{\mathrm{post}}_{\mathrm{ss}}$.
  \item Too small $\to$ terminal condition contaminates the path.
\end{itemize}

\vspace{0.4em}

\textbf{Rule of thumb.} $T$ such that $|\bar K_T - \bar K^{\mathrm{post}}_{\mathrm{ss}}| < 10^{-3}$. We use $T = 200$.

\end{frame}
