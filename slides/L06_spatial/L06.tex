\section[Introduction]{Introduction: A Small-Open-Economy Spatial Model}
% =============================================================

\begin{frame}{Three Locations}
  \begin{itemize}
    \item Three locations $j\in\{1,2,3\}$.
    \item Each produces a costlessly-tradable, perfectly-substitutable numeraire.
    \item Each has its own \textbf{exogenous productivity} $A_j$: $A_1 > A_2 > A_3$.
    \item Local Cobb-Douglas production: $Y_j = A_j K_j^{\alpha} L_j^{1-\alpha}$.
  \end{itemize}
\end{frame}

\begin{frame}{Households}
  \begin{itemize}
    \item Each household supplies one unit of labour inelastically.
    \item \textbf{Idiosyncratic state}: $(b, j)$, wealth $b$ and current location $j$.
    \item No idiosyncratic income shocks. Within $j$, every household earns $w_j$.
  \end{itemize}
\end{frame}

\begin{frame}{Bond Market}
  Households save in a liquid asset, trade with the rest of the world at exogenous interest rate $r$.

  This also pins down capital and wages:
  \begin{align*}
    r + \delta &= \alpha A_j (L_j / K_j)^{1 - \alpha}\\
    w_j &= (1 - \alpha) A_j \left( \frac{\alpha A_j}{r + \delta} \right)^{\alpha / (1 - \alpha)}.
  \end{align*}
\end{frame}

\begin{frame}{Within-Period Stage Chain}
  \begin{enumerate}
    \item \textbf{Migration.}
    \begin{itemize}
      \item Households draw Gumbel taste shocks over destinations
      \item Pick $j'$, pay cost $C[j, j']$, get amenity bonus $\alpha_{j'}$.
    \end{itemize}
    \item \textbf{Wealth update.} At destination $j'$: $b' \gets (1 + r) b + w_{j'}$.
    \item \textbf{Consumption / savings.} Choose next-period wealth.
  \end{enumerate}
  Household block in Stage notation:
  \[
    \text{Migration} \circ \text{WealthChange} \circ \text{ConsumptionSavings}
  \]
\end{frame}

\begin{frame}{Steady State Objects}
  Given parameters, the model should output the \textbf{steady state} values of,
  \begin{itemize}
    \item Population shares $s_j$ across locations.
    \item Wealth distribution $\Lambda(b, j)$.
    \item Aggregate $K_{\text{total}} = \sum_b b\, \Lambda(b, \cdot)$.
    \item Value function $V(b, j)$.
  \end{itemize}
  Note: Prices $r$ and $\{w_j\}_j$ are pinned down before passing to the household block.
\end{frame}

% =============================================================
\section[MigrationStage]{MigrationStage}
% =============================================================

\begin{frame}{Gumbel (AKA Logit) Taste Shocks}
  Each household $h$ in location $i$ draws idiosyncratic \textbf{Gumbel taste shocks} $\varepsilon_{hj} \sim \text{EV1}$ from a Gumbel distribution, one for each destination $j$.

  At origin $i$, the per-destination payoff is
  \begin{align*}
    u_{hij} &= u_{ij} + \eta \varepsilon_{hj},\\
    \text{with}\quad u_{ij} &= \alpha_j -C[i, j] + \Vend(b, j).
  \end{align*}

  The household solves,
  \[
    j'_h = \arg\max_j u_{ij} + \eta\varepsilon_{hj}.
  \]

  Note: A \textbf{higher} value of $\eta$ makes households \textbf{less} sensitive to spatial wage differentials via $\Vend$.
\end{frame}

\begin{frame}{Choice Probabilities}
  With $\varepsilon_{hj} \sim \text{EV1}$, two classical facts:

  \textbf{(a) Choice probabilities are logit:}
  \[
  \Pr(j \mid i, b) = \frac{\exp(u_{ij} / \eta)}{\sum_k \exp(u_{ik} / \eta)}.
  \]

  \textbf{(b) Expected value is given by:}
  \[
  \E\Big[\max_j (u_{ij} + \eta \cdot \varepsilon_{hj})\Big] = \eta \log \sum_k \exp\!\big(u_{ik} / \eta \big).
  \]
\end{frame}

\begin{frame}{Mathematical Definition of \texttt{MigrationStage}}
  \textbf{$V$ Backward:}
  \[
  V_{\text{pre}}(b, i) = \eta \log \sum_j \exp\!\frac{- C[i,j] + \alpha_j + V_{\text{end}}(b, j)}{\eta}.
  \]

  \textbf{$\Lambda$ Forward:}
  \[
  \Lambda_{\text{post}}(b, j) = \sum_i \Pr(j \mid i, b)\, \Lambda_{\text{pre}}(b, i).
  \]
\end{frame}

\section[Steady State]{Steady State}
% =============================================================

\begin{frame}[fragile]{Inner solve}
  The library function \texttt{solve\_steady\_state\_given\_env!}, given \texttt{env}, iterates V backward to fixed point (VFI), iterates $\Lambda$ forward to convergence, then computes moments.

  \begin{lstlisting}
  function solve_household(hh, p; V_init=nothing, Λ_init=nothing)
      env = make_env(hh; r=p.r, w=spatial_wage.(p.r, p.A, p))
      (;V, Λ, moments) = solve_steady_state_given_env!(hh, env; V_init, Λ_init)
      shares = moments.L ./ sum(moments.L)
      return (; V, Λ, env, moments, shares)
  end
  \end{lstlisting}
\end{frame}

\section[$\alpha$ Calibration]{Calibration of $\alpha$}
% =============================================================

\begin{frame}{Outer Loop: Indirect Inference}
  This is an example of how calibration via \textbf{indirect inference} can be thought of as an\\ \textbf{outer loop} around the household block.

  \textbf{Note:} In reality, all parameters would be calibrated together in a joint or nested way. In this lesson, we take all other parameters as given for simplicity.
\end{frame}

\begin{frame}{Calibrating $\alpha$}
  Suppose that in the data we observe shares,
  \[
    s^\text{data} = (0.30, 0.30, 0.40).
  \]
  We want to match these shares in the model-simulated data.

  The structure of the model implies that each steady-state population share $s_j^\text{model}$ is increasing in $\alpha_j$.

  Our strategy is then to update each $\alpha_j$ according to the log-share gap $\log s_j^{\text{data}} - \log s_j^{\text{model}}$.
\end{frame}

\begin{frame}{Empirical Targeted Moments}
  \textbf{Empirical Targeted Moments} (population shares):
  \[
  s^{\text{data}} = (0.30,\ 0.30,\ 0.40).
  \]

  \begin{itemize}
    \item In the data, \emph{40\%} of households live in location 3, even though its wage is \emph{lowest}.
    \item We posit there is an \textbf{unexplained/unobserved portion} of people's desire to live in location 3.
    \item We call this $\alpha_3$: the \textbf{amenity} of location 3.
  \end{itemize}
\end{frame}

\begin{frame}{Calibration Outer Loop}
  \begin{enumerate}
    \item Initialize $\alpha^{(0)} = 0$.
    \item \textbf{Solve the household} at $\alpha^{(k)}$. Get $s^{\text{model}, (k)}$.
    \item Compute gap $g_j = \log s_j^{\text{data}} - \log s_j^{\text{model}, (k)}$.
    \item If $\| g \|_\infty < \text{tol}$, \textbf{stop}.
    \item Update $\alpha^{(k+1)} = \alpha^{(k)} + \texttt{update\_speed} \cdot \eta \cdot g$, normalize $\alpha_1 = 0$.
    \item Repeat.
  \end{enumerate}
\end{frame}

\section[Extensions]{Extensions}
% =============================================================

\begin{frame}{This Version}
  \begin{enumerate}
    \item One consumption good.
    \item Perfect substitutability of goods across locations.
    \item Costless trade in that good.
    \item Free capital flow (one $r$).
  \end{enumerate}

  There are many ways you could enrich this.
\end{frame}

\begin{frame}{Differentiated Goods (Armington)}
  Each location's good is a \emph{distinct variety}; consumers love variety (CES aggregator).
  \begin{itemize}
    \item Trade flows from variety demand.
    \item Local prices $p_j$ differ; real wages depend on the local price index.
    \item Migration responds to \emph{real} wages.
  \end{itemize}
  This is the \textbf{Armington (1969)} foundation of much of modern trade / spatial economics.
\end{frame}

\begin{frame}{Trade Costs}
  Models goods transportation as costly. \textbf{Iceberg costs} $\tau_{ij} \geq 1$ imply you must ship $\tau_{ij}$ units for each 1 to arrive.
  \begin{itemize}
    \item Local prices depend on distance.
    \item Some goods become non-tradable ($\tau \to \infty$): services, housing.
    \item Households consume a mix of tradables and non-tradables.
    \item Migration costs and trade costs interact (closer regions tend to share both).
  \end{itemize}
\end{frame}

\begin{frame}{Restricted Factor Mobility}
  \begin{itemize}
    \item \textbf{Capital is local} (e.g.\ housing). Local capital scarcity raises local rental rates and house prices.
    \item \textbf{Migration cost is sunk and large.} Workers don't equalize wages quickly; spatial wage differentials persist.
    \item \textbf{Skill heterogeneity.} High- and low-skill workers face different migration choices --- as in modern urban literature (Diamond 2016, Moretti 2011).
  \end{itemize}
\end{frame}

\begin{frame}{Implementing Extensions}
  \begin{itemize}
    \item Armington / non-tradables: Additional equations determining the prices in \texttt{env} (outer loop / DAG).
    \item Homeownership: Add housing idiosyncratic state variable (axis) and housing buy/sell stages.
    \item Skill heterogeneity (as in Aiyagari): Add idiosyncratic human capital $z$ state variable (axis) and stage.
  \end{itemize}
\end{frame}
