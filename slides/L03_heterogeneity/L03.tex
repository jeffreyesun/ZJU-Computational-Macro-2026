\section{Policy Function}
% =============================================================

\begin{frame}{Policy Function: $c^\star$}
  The policy function is the \textbf{argmax} version of the \textbf{max} Bellman operator:
  \begin{align*}
    \mathcal{T}v(b)        &\;=\; \max_{c \in [0, b]} \; u(c) + \beta \, v\!\big(R(b - c) + z\big) \tag{Bellman Operator} \\[0.4em]
    c^\star(b)  &\;=\; \arg\max_{c \in [0, b]} \; u(c) + \beta \, v\!\big(R(b - c) + z\big) \tag{Policy Function}
  \end{align*}
  \begin{itemize}
    \item Once the true $V$ is solved, $\arg\max$ gives you the optimal policy $c^\star$.
    \item More deeply, any $v$ gives you some policy function.
    \begin{itemize}
      \item $v$ can be thought of as representing beliefs about the value of being in various states.
    \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{Marginal Propensity to Consume (MPC)}
  The slope $\partial c^\star / \partial b$ is the \textbf{Marginal Propensity to Consume (MPC)}
  
  Generally (theoretically and empirically) higher for poorer households, low for rich.

  Can be thought of as a function of how \textbf{constrained} households are.
\end{frame}

\section{Conventions}
% =============================================================

\begin{frame}{Conventions}
  \begin{itemize}
    \item Now that we have some intuition, let's set up some conventions that will pay off later.
    \item Think of the state variable $z$ as \textbf{human capital},following a \textbf{Markov process $\Pi_z$}
    \item Period income is $zw$, where $w$ is the wage level. For now, we'll normalize $w = 1$.
    \item The \textbf{state} is now $(b, z)$.
    \item On computer, each $V$ is now a \textbf{matrix}. Rows are wealth $b$, columns are human capital $z$.
  \end{itemize}
\end{frame}

\begin{frame}{Stages}  
  Each \textbf{model feature} can be thought of as a \textbf{Stage}:
  \begin{enumerate}
    \item Income Shock
    \item Income
    \item Consumption-Saving
  \end{enumerate}
  Each \textbf{Stage} is a \textbf{simple sub-period}.
  \begin{itemize}
    \item More on this later, but \textbf{functional composability} of stages is analogous to the \textbf{additive composability} of continuous-time household models.
    \item This can allow us to build \textbf{complex} models by composing \textbf{Stages}.
  \end{itemize}
\end{frame}

\section{Stages}

\begin{frame}{Stage 1 --- Income Shock}
  \begin{itemize}
    \item $(\Pi_z)_{ij} = \Pr(z_t = j \mid z^{\mathrm{end}}_{t-1} = i)$, row-stochastic.
    \item \textbf{Backward} (value functions).
  \end{itemize}
  \[
    V^{\mathrm{start}} \;=\; V^{\mathrm{pre\_inc}} \, \Pi_z^\top.
  \]
  Each $(b^{\mathrm{end}}, z^{\mathrm{end}}_{t-1})$ entry of $V^{\mathrm{start}}$ is an expectation over $z_t$ given $z^{\mathrm{end}}_{t-1}$.

  (Note: To see why matrix multiplication can represent backwards-iteration of $V$ through a Markov chain, see standard references on Markov chains.)
\end{frame}

\begin{frame}{Stage 2 --- Income}
  This stage just represents the \textbf{receipt} of income,
  \[
    b_t = R \, b^{\mathrm{end}}_{t-1} + z_t,
  \]
  effectively a change-of-variables.

  Explicitly, the backwards step is,
  \begin{align*}
    V^{\mathrm{pre\_inc}}(b^{\mathrm{end}}_{t-1}, z)
    \; &= \; V^{\mathrm{pre\_cons}}\!\big(R\, b^{\mathrm{end}}_{t-1} + z,\; z\big).
  \end{align*}

  In the example code, we do this by \textbf{reinterpolating} $V^{\mathrm{pre\_cons}}$ at the new points $R\, b^{\mathrm{end}}_{t-1} + z$.
\end{frame}

\begin{frame}{Stage 3 --- Consumption-Saving}
  This stage represents the argmax step,
  \[
    b^{\mathrm{end}}_t \;=\; b_t - c^\star(b_t, z_t).
  \]
  Explicitly, the backwards step is,
  \[
    V^{\mathrm{pre\_cons}}(b, z) \;=\; \max_{b^{\mathrm{end}} \in b_{\mathrm{grid}}} u(b - b^{\mathrm{end}}) + \beta V^{\mathrm{end}}(b^{\mathrm{end}}, z).
  \]
  
  Here, we fold in the time discounting by $\beta$ in the consumption-saving stage, but in general it can be thought of as really a separate stage.
\end{frame}

\begin{frame}[fragile]{Backwards Iteration}
  \begin{lstlisting}
"Iterate V_end backwards by one full period."
function bellman_operator(V_end, params)
    (;β) = params

    # Stage 3
    V_post_inc = consumption_saving_backward(V_end, params)

    # Stage 2
    V_post_inc_shock = income_backward(V_post_inc, params)

    # Stage 1
    V_start = income_shock_backward(V_post_inc_shock, params)

    # Stage 0
    V_end_new = β .* V_start

    return V_end_new
end
  \end{lstlisting}
  Backwards iteration of $V$ is just \textbf{functional composition} of the backwards-iteration functions of each stage.
\end{frame}

\section{Simulating a Population}
% =============================================================

\begin{frame}{Heterogeneity for (Almost) Free}
  \begin{itemize}
    \item We have been thinking of $V$ and $c^*$ as the value and policy of a single household at different \textbf{possible} states.
    \item But we can also think of $V$ and $c^*$ as the value and policy of \textbf{contemporaneous} households in a \textbf{population} with different idiosyncratic states.
    \item Since we have already computed $V$ on the whole grid, this lets us add heterogeneity very cheaply.
    \item We just need to think about how to \textbf{represent} and \textbf{simulate} a population of households.
  \end{itemize}
\end{frame}

\begin{frame}{A Population is a Vector (or Matrix)}
  We can represent the \emph{population} as a discrete distribution $\Lambda$ over the wealth grid:
  \[
    \Lambda_i \;=\; \text{share of households at } b_i.
  \]
  That is, just like $V$, it is a vector (or a matrix) over the discretized state space.

  (Note: Matrices are vectors.)
\end{frame}

\begin{frame}{Aggregate Moments}
  \begin{itemize}
    \item Aggregate welfare is then an inner product: $\bar{V} = \sum_i V(b_i) \, \Lambda_i$. Same for $\bar{C}$, $\bar{K}$, etc.
    \item Indeed, \textbf{any} moment of the distribution (e.g. aggregate capital, consumption, etc.) can be thought of as an inner product against $\Lambda$.
  \end{itemize}
\end{frame}

\begin{frame}{Forward and Backward, All Three Stages}
  \footnotesize
  \setlength{\tabcolsep}{6pt}
  \renewcommand{\arraystretch}{1.45}
  \begin{tabular}{@{}l l l@{}}
  \toprule
  & \textbf{Backward} (value) & \textbf{Forward} (distribution) \\
  \midrule
  \textbf{Stage 1}    & $V^{\mathrm{start}} = V^{\mathrm{pre\_inc}}\, \Pi_z^\top$
                      & $\Lambda^{\mathrm{post}} = \Lambda^{\mathrm{pre}}\, \Pi_z$ \\
  \textbf{Stage 2}    & $V^{\mathrm{pre\_inc}}(b^{\mathrm{end}}, z) = V^\text{pre\_cons}(R\, b^{\mathrm{end}} + z,\, z)$
                      & $(b^{\mathrm{end}}, z) \to (R\, b^{\mathrm{end}} + z,\, z)$ \\
  \textbf{Stage 3}    & $V^\text{pre\_cons}(b, z) = \max\limits_{b^{\mathrm{end}}} u(b - b^{\mathrm{end}}) + V^{\mathrm{end}}(b^{\mathrm{end}}, z)$
                      & $(b, z) \to \big(b - c^\star(b, z),\, z\big)$ \\
  \midrule
  \textbf{Dynamic Boundary Condition}   & $V^{\mathrm{end}_t} = \beta\, V^{\mathrm{start}}_{t+1}$
                      & $\Lambda^\text{start}_t = \Lambda^\text{end}_{t+1}$ \\
  \textbf{Steady State}   & $V^{\mathrm{end}} = \beta\, V^{\mathrm{start}}$
                      & $\Lambda^\text{start} = \Lambda^\text{end}$ \\
  \bottomrule
  \end{tabular}

  \vspace{0.8em}

  \footnotesize
  Backward: $V^{\mathrm{end}} \to V^\text{pre\_cons} \to V^{\mathrm{pre\_inc}} \to V^{\mathrm{start}}$.

  Forward: $\Lambda^\text{start} \to \Lambda^\text{pre\_inc} \to \Lambda^\text{pre\_cons} \to \Lambda^\text{end}$.
\end{frame}

\begin{frame}[fragile]{Stage 1: Markov Income Shock}
  Forward:
  \begin{lstlisting}
# Λ is an N_b × N_z matrix: rows index wealth, columns index z state.
# Markov along the z dimension is a single matrix multiplication.
income_shock_forward(Λ, Π_z) = Λ * Π_z
  \end{lstlisting}

  \begin{itemize}
    \item The mass at $(b_i, z_j)$ flows to $(b_i, z_{j'})$ with weight $(\Pi_z)_{j j'}$.
    \item Markov along one dimension is just a matrix multiplication.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{\texttt{simulate\_population\_forward}}
  \begin{lstlisting}
"Simulate population forward through all three Stages: Λ → income shock → income → consumption saving."
function simulate_population_forward(Λ, b_inds_new, params)
    # Stage 1
    Λ = income_shock_forward(Λ, params.Π_z)

    # Stage 2
    Λ = income_forward(Λ, params)

    # Stage 3
    Λ = consumption_saving_forward(Λ, b_inds_new, params)

    return Λ
end
  \end{lstlisting}
  Forward simulation is just \textbf{functional composition} of the forward-simulation functions of each stage.
\end{frame}


\begin{frame}{Solving for Steady State}
  \begin{enumerate}
    \item Guess $v$ and $\Lambda$.
    \item Iterate $v$ backwards until it converges to $V$.
    \item Simulate $\Lambda$ forward until it converges to its steady state.
  \end{enumerate}
  (Note: The conditions under which $\Lambda$ converges are subtler than for $V$.)
\end{frame}
