% =============================================================
\section{Policy Function / 策略函数}
% =============================================================

\begin{frame}{Policy Function: $c^\star$ \\ 策略函数 $c^\star$}
  \begin{align*}
    V(b)        &\;=\; \max_{c \in [0, b]} \; u(c) + \beta \, V\!\big(R(b - c) + w\big) \tag{Bellman Operator} \\[0.4em]
    c^\star(b)  &\;=\; \arg\max_{c \in [0, b]} \; u(c) + \beta \, V\!\big(R(b - c) + w\big) \tag{Policy Function}
  \end{align*}

  \begin{columns}[T]
  \begin{column}{0.46\textwidth}
  Same right-hand side; the only difference is $\max$ vs.\ $\arg\max$.

  \vspace{0.4em}

  Once $V$ is solved, one $\arg\max$ pass reads off the policy $c^\star$.
  \end{column}
  \hfill
  \begin{column}{0.46\textwidth}
  右端完全相同，区别仅在 $\max$ 与 $\arg\max$。

  \vspace{0.4em}

  只要求出 $V$，再做一次 $\arg\max$ 就能读出策略 $c^\star$。
  \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Features of $c^\star$ \\ $c^\star$ 的形态}
  \begin{columns}[T]
  \begin{column}{0.46\textwidth}
  $c^\star(b)$ is \textbf{increasing} (in this and most models): richer households consume more.

  \vspace{0.4em}

  The slope $\partial c^\star / \partial b$ is the \textbf{Marginal Propensity to Consume (MPC)}: high for poorer households, low for rich.
  \end{column}
  \hfill
  \begin{column}{0.46\textwidth}
  $c^\star(b)$ \textbf{单调递增}（本模型与大多数模型皆然）：财富越多消费越多。

  \vspace{0.4em}

  斜率 $\partial c^\star / \partial b$ 就是\textbf{边际消费倾向 (MPC)}：贫者高，富者低。
  \end{column}
  \end{columns}
\end{frame}

\section{Timing / 时序}
% =============================================================

\begin{frame}{Income Shocks \\ 收入冲击}
  \begin{columns}[T]
  \begin{column}{0.46\textwidth}
  We now add \textbf{income} as a \textbf{state variable}, following a \textbf{Markov process $\Pi$}.

  \vspace{0.4em}

  The \textbf{state} is now $(b, w)$.

  \vspace{0.4em}

  On the computer, each $V$ is now a \textbf{matrix}. Rows are wealth $b$, columns are income $w$.
  \end{column}
  \hfill
  \begin{column}{0.46\textwidth}
  现在把\textbf{收入}加为一个\textbf{状态变量}，并令其服从一个 \textbf{马尔可夫过程 $\Pi$}。

  \vspace{0.4em}

  \textbf{状态}变为 $(b, w)$。

  \vspace{0.4em}

  在计算机里，每个 $V$ 现在是一个\textbf{矩阵}。行对应财富 $b$，列对应收入 $w$。
  \end{column}
  \end{columns}
\end{frame}

    \item Income Shock
    \item Income
    \item Consumption-Saving
  \end{enumerate}

  \vspace{0.6em}

  Each stage is a \textbf{simple sub-period}.
  \end{column}

  \begin{column}{0.46\textwidth}
  可以把每个\textbf{模型特征}视为一个\textbf{阶段}：
  \begin{enumerate}
    \item 收入冲击
    \item 收入

  \vspace{0.6em}

  每个阶段都是一个\textbf{简单的子周期}。
  \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Stage 1 --- Income Shock \\ 第一阶段：收入冲击}
$\Pi_{ij} = \Pr(w_t = j \mid w^{\mathrm{end}}_{t-1} = i)$, row-stochastic.

\vspace{0.8em}

\begin{columns}[T]
\begin{column}{0.46\textwidth}
\textbf{Backward} (value functions).
\[
  V^{\mathrm{start}} \;=\; V^{\mathrm{pre\_inc}} \, \Pi^\top.
\]

\vspace{0.6em}

Each $(b^{\mathrm{end}}, w^{\mathrm{end}}_{t-1})$ entry of $V^{\mathrm{start}}$ is an expectation over $w_t$ given $w^{\mathrm{end}}_{t-1}$.
\end{column}
\hfill
\begin{column}{0.46\textwidth}
\textbf{后向}（价值函数）。
\[
  V^{\mathrm{start}} \;=\; V^{\mathrm{pre\_inc}} \, \Pi^\top.
\]

\vspace{0.6em}

$V^{\mathrm{start}}$ 中每个 $(b^{\mathrm{end}}, w^{\mathrm{end}}_{t-1})$ 项是给定 $w^{\mathrm{end}}_{t-1}$ 时对 $w_t$ 取的期望。
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Why Matrix Multiplication? \\ 为什么是矩阵乘法？}
\begin{columns}[T]
\begin{column}{0.46\textwidth}
Hold $b^{\mathrm{end}}$ fixed. The values
\[
  v_j \;\equiv\; V^{\mathrm{pre\_inc}}(b^{\mathrm{end}}, w_j), \qquad j = 1, \ldots, N_w
\]
form a length-$N_w$ vector.

\vspace{0.5em}

The Markov expectation is, by definition,
\[
  V^{\mathrm{start}}(b^{\mathrm{end}}, w_i) \;=\; \sum_j \Pi_{ij}\, v_j.
\]

\vspace{0.4em}

That is exactly the $i$-th component of $\Pi\, v$ — a matrix--vector product.
\end{column}
\hfill
\begin{column}{0.46\textwidth}
固定 $b^{\mathrm{end}}$。值
\[
  v_j \;\equiv\; V^{\mathrm{pre\_inc}}(b^{\mathrm{end}}, w_j), \qquad j = 1, \ldots, N_w
\]
构成一个长度 $N_w$ 的向量。

\vspace{0.5em}

由马尔可夫期望的定义，
\[
  V^{\mathrm{start}}(b^{\mathrm{end}}, w_i) \;=\; \sum_j \Pi_{ij}\, v_j.
\]

\vspace{0.4em}

这正是 $\Pi\, v$ 的第 $i$ 个分量——一次矩阵--向量乘积。
\end{column}
\end{columns}
\end{frame}

\begin{frame}{All Wealth Rows at Once \\ 所有财富同时算}
\begin{columns}[T]
\begin{column}{0.46\textwidth}
Stack the per-$b^{\mathrm{end}}$ vectors: $V^{\mathrm{pre\_inc}}$ is an $N_b \times N_w$ matrix.

\vspace{0.5em}

The Markov transition acts \emph{identically} on each row, so the same trick applies in batch:
\[
  V^{\mathrm{start}} \;=\; V^{\mathrm{pre\_inc}} \, \Pi^\top.
\]

\vspace{0.4em}

One matrix multiply updates every wealth row.

\vspace{0.5em}

\textbf{Markov along one dimension is a single matrix multiplication.}
\end{column}
\hfill
\begin{column}{0.46\textwidth}
将每个 $b^{\mathrm{end}}$ 对应的向量堆叠：$V^{\mathrm{pre\_inc}}$ 是 $N_b \times N_w$ 矩阵。

\vspace{0.5em}

马尔可夫转移对每一行的作用\emph{完全相同}，于是可以批量处理：
\[
  V^{\mathrm{start}} \;=\; V^{\mathrm{pre\_inc}} \, \Pi^\top.
\]

\vspace{0.4em}

一次矩阵乘法即可更新所有财富行。

\vspace{0.5em}

\textbf{沿一维做马尔可夫就是一次矩阵乘法。}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Stage 2 --- Income \\ 第二阶段：收入}
$b_t \;=\; R \, b^{\mathrm{end}}_{t-1} + w_t$.

\vspace{0.8em}

\begin{columns}[T]
\begin{column}{0.46\textwidth}
\textbf{Backward} (value functions).
\begin{align*}
  V^{\mathrm{pre\_inc}}(b^{\mathrm{end}_{t-1}}, w)
   \; &= \; V\!\big(R\, b^{\mathrm{end}_{t-1}} + w,\; w\big)\\
  V^{\mathrm{pre\_inc}}((b-w)/R, w)
   \; &= \; V\!\big(b, w\big)
\end{align*}
\end{column}
\hfill
\begin{column}{0.46\textwidth}
\textbf{后向}（价值函数）。
\begin{align*}
  V^{\mathrm{pre\_inc}}(b^{\mathrm{end}_{t-1}}, w)
   \; &= \; V\!\big(R\, b^{\mathrm{end}_{t-1}} + w,\; w\big)\\
  V^{\mathrm{pre\_inc}}((b-w)/R, w)
   \; &= \; V\!\big(b, w\big)
\end{align*}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Stage 3 --- Consumption-Saving \\ 第三阶段：消费--储蓄}
$b^{\mathrm{end}}_t \;=\; b_t - c^\star(b_t, w_t)$.

\vspace{0.8em}

\begin{columns}[T]
\begin{column}{0.46\textwidth}
\textbf{Backward} (value functions). Grid search over $b^{\mathrm{end}}$:
\[
  V(b, w) \;=\; \max_{b^{\mathrm{end}} \in b_{\mathrm{grid}}} u(b - b^{\mathrm{end}}) + V^{\mathrm{end}}(b^{\mathrm{end}}, w).
\]

\vspace{0.6em}

\textit{Period boundary:} $V^{\mathrm{end}} = \beta\, V^{\mathrm{start}}$.
\end{column}
\hfill
\begin{column}{0.46\textwidth}
\textbf{后向}（价值函数）。在 $b^{\mathrm{end}}$ 上做网格搜索：
\[
  V(b, w) \;=\; \max_{b^{\mathrm{end}} \in b_{\mathrm{grid}}} u(b - b^{\mathrm{end}}) + V^{\mathrm{end}}(b^{\mathrm{end}}, w).
\]

\vspace{0.6em}

\textit{期界过渡：} $V^{\mathrm{end}} = \beta\, V^{\mathrm{start}}$。
\end{column}
\end{columns}
\end{frame}

\section{A Population is a Vector / 群体即向量}
% =============================================================

\begin{frame}{A Population is a Vector \\ 群体即向量}
\begin{columns}[T]
\begin{column}{0.46\textwidth}
The same $c^\star$ is the rule for every household at every wealth level.

\vspace{0.4em}

So we can think of a \emph{population} as a distribution $\lambda$ on the wealth grid:
\[
  \lambda_i \;=\; \text{share of households at } b_i.
\]

\vspace{0.4em}

A population is just a vector — same shape as $V$, same shape as $c^\star$.
\end{column}
\hfill
\begin{column}{0.46\textwidth}
对任何家庭、任何财富水平，$c^\star$ 都给出同一条规则。

\vspace{0.4em}

于是可以把\emph{群体}视为财富网格上的一个分布 $\lambda$：
\[
  \lambda_i \;=\; \text{处于 } b_i \text{ 的家庭占比}.
\]

\vspace{0.4em}

群体就是一个向量——与 $V$、$c^\star$ 同形状。
\end{column}
\end{columns}
\end{frame}

\begin{frame}{$V$ Is Also Cross-Sectional Welfare \\ $V$ 也是横截面福利}
\begin{columns}[T]
\begin{column}{0.46\textwidth}
$V(b_i)$ has two readings:
\begin{itemize}
  \item the value to \emph{one} household if its wealth were $b_i$;
  \item the current welfare of the households who actually have $b_i$.
\end{itemize}

\vspace{0.4em}

Aggregate welfare is then an inner product: $\bar{V} = \sum_i V(b_i) \, \lambda_i$. Same for $\bar{C}$, $\bar{K}$, etc.

\vspace{0.5em}

\textbf{One $V$, two uses.} Solve once, get aggregates for free.
\end{column}
\hfill
\begin{column}{0.46\textwidth}
$V(b_i)$ 有两种读法：
\begin{itemize}
  \item 设想\emph{一个}家庭的财富是 $b_i$，则其价值；
  \item 实际财富为 $b_i$ 的那群家庭当下的福利。
\end{itemize}

\vspace{0.4em}

总福利就是一次内积：$\bar{V} = \sum_i V(b_i) \, \lambda_i$。$\bar{C}$、$\bar{K}$ 等类推。

\vspace{0.5em}

\textbf{一个 $V$，两种用途。} 解一次，所有总量都送上门。
\end{column}
\end{columns}
\end{frame}

\section{Simulating a Population / 群体模拟}
% =============================================================

\begin{frame}{Forward and Backward, All Three Stages \\ 前向与后向：三阶段汇总}
\footnotesize
\setlength{\tabcolsep}{6pt}
\renewcommand{\arraystretch}{1.45}
\begin{tabular}{@{}l l l@{}}
\toprule
& \textbf{Backward} (value) & \textbf{Forward} (distribution) \\
\midrule
\textbf{Stage 1}    & $V^{\mathrm{start}} = V^{\mathrm{pre\_inc}}\, \Pi^\top$
                    & $\lambda^{\mathrm{post}} = \lambda^{\mathrm{pre}}\, \Pi$ \\
\textbf{Stage 2}    & $V^{\mathrm{pre\_inc}}(b^{\mathrm{end}}, w) = V(R\, b^{\mathrm{end}} + w,\, w)$
                    & $(b^{\mathrm{end}}, w) \to (R\, b^{\mathrm{end}} + w,\, w)$ \\
\textbf{Stage 3}    & $V(b, w) = \max\limits_{b^{\mathrm{end}}} u(b - b^{\mathrm{end}}) + V^{\mathrm{end}}(b^{\mathrm{end}}, w)$
                    & $(b, w) \to \big(b - c^\star(b, w),\, w\big)$ \\
\midrule
\textbf{Boundary}   & $V^{\mathrm{end}} = \beta\, V^{\mathrm{start}}$
                    & --- \\
\bottomrule
\end{tabular}

\vspace{0.8em}

\begin{columns}[T]
\begin{column}{0.46\textwidth}
\footnotesize Backward: VFI iterates $V^{\mathrm{end}} \to V \to V^{\mathrm{pre\_inc}} \to V^{\mathrm{start}}$, then loops. Forward: the population's $\lambda$ traces the same three stages in time order.
\end{column}
\hfill
\begin{column}{0.46\textwidth}
\footnotesize 后向：VFI 沿 $V^{\mathrm{end}} \to V \to V^{\mathrm{pre\_inc}} \to V^{\mathrm{start}}$ 迭代后再循环。前向：群体 $\lambda$ 按时间顺序经历同样的三个阶段。
\end{column}
\end{columns}
\end{frame}

\begin{frame}[fragile]{One Markov Step Along the Income Dimension \\ 沿收入维做一步马尔可夫}
\begin{lstlisting}
# λ is an N_b × N_w matrix: rows index wealth, columns index income state
# applying Π along the income dimension is a single matrix multiplication:
λ_new = λ * Π
\end{lstlisting}

\begin{columns}[T]
\begin{column}{0.46\textwidth}
The mass at $(b_i, w_j)$ flows out to $(b_i, w_{j'})$ with weight $\Pi_{j j'}$.

\vspace{0.4em}

Wealth doesn't move in this step — that's Stages 2 and 3.

\vspace{0.4em}

\textbf{Markov along one dimension is just a matrix multiplication.} We'll use this trick repeatedly in later lectures.
\end{column}
\hfill
\begin{column}{0.46\textwidth}
$(b_i, w_j)$ 处的质量按权重 $\Pi_{j j'}$ 流向 $(b_i, w_{j'})$。

\vspace{0.4em}

财富在这一步不动——那是第二、第三阶段的事。

\vspace{0.4em}

\textbf{沿一维做马尔可夫就是一次矩阵乘法。} 这个套路后面会反复用到。
\end{column}
\end{columns}
\end{frame}

\begin{frame}[fragile]{\texttt{simulate\_population} / 群体模拟}
\begin{lstlisting}
function simulate_population(c_inds, params, λ_0; T = 200)
    λ_path, λ = [copy(λ_0)], copy(λ_0)
    for t in 1:T
        λ = income_shock_forward(λ, params.Π)             # Stage 1
        λ = income_forward(λ, params)                      # Stage 2
        λ = consumption_saving_forward(λ, c_inds, params)  # Stage 3
        push!(λ_path, copy(λ))
    end
    return λ_path
end
\end{lstlisting}

\footnotesize
\begin{columns}[T]
\begin{column}{0.46\textwidth}
One forward step is just the three named stage functions composed in order. Stage 1 is a matrix multiply on the income dimension; Stages 2 and 3 are pure index moves --- cash-on-hand assembly, then policy lookup. No lotteries.
\end{column}
\hfill
\begin{column}{0.46\textwidth}
一次前向就是按顺序复合三个阶段函数。第一阶段是收入维上的矩阵乘法；第二、第三阶段都是纯下标移动——先装配现金，再查策略。不用抽奖。
\end{column}
\end{columns}
\end{frame}

\begin{frame}{The Long-Run Distribution \\ 长期分布}
\begin{columns}[T]
\begin{column}{0.46\textwidth}
Apply the forward step over and over. After enough periods the distribution $\lambda$ stops changing.

\vspace{0.5em}

The limit distribution $\lambda^\star$ describes the long-run population. Once we have it, aggregates are just inner products with $b$, $V$, $c^\star$, etc.
\end{column}
\hfill
\begin{column}{0.46\textwidth}
反复施加一步前向算子。期数足够多后，$\lambda$ 不再变化。

\vspace{0.5em}

极限分布 $\lambda^\star$ 描述了群体的长期形态。有了它，总量就是与 $b$、$V$、$c^\star$ 等做内积。
\end{column}
\end{columns}
\end{frame}
