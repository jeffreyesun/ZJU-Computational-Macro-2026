\begin{frame}{Model Structure}
  These slides: solving heterogeneous-agent (HA) GE models with and without aggregate shocks in Julia.

  We cover discrete-time GE models with the following form:
  \begin{enumerate}
    \item A forward-looking household block made up of several \textbf{Stages}
    \item A set of equations describing the rest of the economy, such as firms, government, etc., typically represented as a \textbf{directed acyclic graph} (DAG) of functions.
    \item A set of \textbf{outer loops} solving, e.g., equilibrium, steady state, calibration, transition, or global solution
  \end{enumerate}

\section{Motivation}
% =============================================================

\begin{frame}{Heterogeneous Agents: Motivation 1}
  \begin{itemize}
    \item Empirical average marginal propensity to consume (MPC) high. Average savings \textbf{also} high
    \item Typical representative-agent model cannot easily explain this.
    \item However, if poor households have high MPC and low saving; and rich households have low MPC/high saving; then a \textbf{heterogeneous-agent} model can explain.
  \end{itemize}
\end{frame}

\begin{frame}{Heterogeneous Agents: Motivation 2}
\begin{itemize}
  \item Change in (e.g. interest rate) policy will affect different people \textbf{differently}
  \item A policy will have different \textbf{aggregate} effects depending on who it targets.
  \item We care about the distributional and aggregate consequences of policy.
\end{itemize}

\end{frame}

\section{Household Problem: Two Period}
% =============================================================

\begin{frame}{Microfoundation: One Household}
  \begin{itemize}
    \item Macroeconomics is the study of how individual behaviour shapes aggregate outcomes.
    \item To understand the origins of macro outcomes, need to study the sources of individual behaviour.
  \end{itemize}
\end{frame}

\begin{frame}{Two Period Model}
  Household starts with \textbf{wealth} $b_0$ in Period 0. In Period 1, they receive income $z_1$.
  \[
    \max_{c_0, c_1 \ge 0} \; u(c_0) + \beta u(c_1)
  \]
  \[
    \text{s.t.}\quad b_0 + \frac{z_1}{R} \geq c_0 + \frac{c_1}{R}.
  \]
  \begin{itemize}
    \item If you save \$1 today, you can spend \$$R$ tomorrow, but utility discounted by $\beta$.
    \item Lagrangian: $\mathcal{L} = u(c_0) + \beta u(c_1) - \lambda\p{c_0 + \frac{c_1}{R} - b_0 - \frac{z_1}{R}}$.
  \end{itemize}
\end{frame}

\begin{frame}{Euler Equation}
  Lagrangian FOCs:
  \begin{align*}
    u'(c_0) &= \lambda, \\
    \beta u'(c_1) &= \frac{\lambda}{R}.
  \end{align*}
  Rearranging,
  \[
    \boxed{\frac{u'(c_0)}{u'(c_1)} = \beta R.}
  \]
  At optimum, \textbf{indifferent} between consuming or saving additional marginal unit today for $R$ units tomorrow.
\end{frame}

\begin{frame}{Example: Analytical Solution with Log Utility}
  Let $u(c) = \log c$. Then, $c_1 = \beta R c_0$, and
  \[
    c_0^\star = \frac{b_0 + \frac{z_1}{R}}{1+\beta},
    \qquad
    c_1^\star = \frac{\beta R \, b_0 + \frac{z_1}{R}}{1+\beta}.
  \]
  With $\beta = 0.96$, $R = 1.04$, $b_0 = 10$, $z_1 = 2$:
  \[
    c_0^\star \approx 6.08, \;\; c_1^\star \approx 6.07.
  \]
  There is \textbf{consumption smoothing}. Income is bumpy at $(10, 2)$. Consumption is smoother at $(6.08, 6.07)$.
\end{frame}

\begin{frame}{Policy Function}
  The \textbf{Policy Function} represents optimal period-0 consumption in terms of \textbf{initial wealth} $b_0$,
  \[
    c_0^\star(b_0) = \frac{b_0 + z_1/R}{1+\beta}.
  \]
  This way, we have a model of household behaviour, and how \textbf{household behaviour} arises.
\end{frame}

\begin{frame}{CRRA Preferences}
  A standard assumption is constant relative risk aversion (CRRA) utility,
  \[
    u(c) = \begin{cases}
      \frac{c^{1-\gamma}-1}{1-\gamma}, & \gamma > 0, \gamma \neq 1, \\
      \log c, & \gamma = 1.
    \end{cases}
  \]
  Utility \textbf{increasing} in $c$,
  \[
    \frac{\partial u(c)}{\partial c} = c^{-\gamma} > 0.
  \]
  Utility is \textbf{concave}: Marginal utility \textbf{decreasing} in $c$,
  \[
    \frac{\partial^2 u(c)}{\partial c^2} = -\gamma c^{-\gamma-1} < 0.
  \]
\end{frame}

\section{Household Problem: $T$ Periods}

\begin{frame}{$T$ Period Model}
  Lifetime utility is given by,
  \[
    \max_{c_0, \ldots, c_{T-1}} \sum_{t=0}^{T-1} \beta^t u(c_t),
    \quad b_{t+1} = R(b_t - c_t) + z_{t+1}.
  \]
  Solving Lagrangian gives us $T-1$ Euler equations,
  \[
    u'(c_t) = \beta R \, u'(c_{t+1}).
  \]
  If future income is perfectly foreseen, can still solve pretty easily.

  But what if income is stochastic?
\end{frame}

\begin{frame}{Stochastic Income}
  \begin{itemize}
    \item Suppose $\{z_t\}$ follows a Markov chain with transition matrix $\Pi$
    \item Given today's $z_t$, we know the \emph{conditional distribution} of $z_{t+1}$, but not the realized value
    \item In each period $t$, household chooses $c_t$ without knowing $z_{t+1}$
  \end{itemize}
\end{frame}

\begin{frame}{Stochastic Euler Equation}
  With $T = 2$, we can substitute,
  \[
    c_1 = R(b_0 - c_0) + z_1.
  \]
  The problem is just,
  \[
    \max_{c_0} u(c_0) + \beta \E\left[\, u\left(R(b_0 - c_0) + z_1\right) \right],
  \]
  with FOC with respect to $c_0$,
  \[
    \boxed{\;u'(c_0) = \beta R \, \E\left[ u'(c_1) \right].}
  \]
  The Euler equation now contains an expectation over \emph{uncertain} $c_1$.
\end{frame}

\begin{frame}{Infinite-Horizon Problem ($T = \infty$)}
  Believe it or not, the problem is actually a little easier if we take $T\to\infty$:
  \begin{align*}
    \max \E_0 & \ \sum_{t=0}^{\infty} \beta^t u(c_t)\\
    \text{s.t.}\quad b_{t+1} &= R(b_t - c_t) + z_{t+1}.
  \end{align*}
  But what are we taking $\max$ over? Formally, we need to find a whole sequence of \textbf{policy functions},
  \[
    c_t = c_t(z_0, \ldots, z_t) \quad \forall t,
  \]
  but that seems almost impossible. Luckily, we can make the problem easier by reformulating it.

  (Note: I'm skipping the issue of the transversality condition.)
\end{frame}

\section{Recursive Form}
% =============================================================

\begin{frame}{Richard Bellman's First Observation}
  At each time $t$, household only makes one choice, then waits,

  \[
    \max_{c_t} = u(c_t) + \beta \E_t\left[ \max \E_{t+1}\sum_{s=t+1}^\infty \beta^t u(c_s) \right]
  \]
  \[
    s.t. \quad b_{s+1} = R(b_s - c_s) + z_{s+1} \quad\forall s \ge t.
  \]

  Importantly, the problem at time $t$ looks \textbf{almost} identical to the problem at time $t+1$.

  The only possible differences are $b_t$ and $z_t$. These are the \textbf{state variables.}
\end{frame}

\begin{frame}{State Variables}
  \begin{itemize}
    \item Informally, \textbf{state variables} are those variables which make time periods (and decisions) meaningfully different from each other.
    \item Here, the state variables are wealth $b_t$ and income $z_t$. Given these, a household's future conditional expected future utilities and optimal decisions are uniquely determined.
  \end{itemize}
\end{frame}

\begin{frame}{Bellman's Second and Third Observations}
  Bellman also realized two more things:
  \begin{enumerate}
    \item[2.] You can express the first insight as an equation about a \textbf{value function} $V$.
    \item[3.] If you know this \textbf{value function}, you can compute the \textbf{policy function} and (numerically) solve the entire model.
  \end{enumerate}
  In computer science terms, this is an example of \textbf{caching}.
\end{frame}

\begin{frame}{Value Function and Bellman Equation}
  Define the \emph{value function} $V(b, z)$ as the \textbf{net present value} of future utility given state $(b, z)$.
  \begin{align*}
    V(b, z) &= \max_{c\in[0,b]} \mathbb{E}_0 \sum_{s=0}^\infty \beta^{s} u(c_s)\\
    \text{s.t.} \quad b_{s+1} &= R(b_s - c_s) + z_{s+1}\ \forall s.
  \end{align*}
  $\quad$(Given that all future choices will also be made optimally.)

  Then $V$ satisfies the \textbf{Bellman equation}
  \[
    \boxed{V(b, z) = \max_{c \in [0, b]} \; u(c) + \beta \, \E\!\left[\, V(b', z') \,\big|\, z \,\right]}
  \]
  s.t. $b' = R(b-c) + z'$
\end{frame}

\begin{frame}{The Value Function Is All You Need}
  If we know $V$, then we can compute the \textbf{policy function} and solve the whole (household) model,
  \[
  c^\star(b, z) = \arg\max_{c \in [0, b]} u(c) + \beta \, \E\!\left[\, V(R(b-c) + z,\, z') \mid z \,\right].
  \]
\end{frame}

\begin{frame}{How do we find $V$?}
  \begin{itemize}
    \item We still have a problem, though. We don't just need to find a number. We need to find an entire \textbf{function} $V$.
    \item We can do this via ``Value Function Iteration'' (VFI).
    \item To understand this: First imagine that we guess a value function, and we get it \textbf{wrong}.
  \end{itemize}

\end{frame}

\begin{frame}{The Bellman operator on arbitrary $v$}
  Suppose that we guess a \textbf{function} $v$. Even though $v$ might be \textbf{wrong}, it would still predict some policy,
  \[
    c^*(b, z; v) = \arg\max_{c \in [0, b]} u(c) + \beta \, \E\!\left[\, v(R(b-c) + z',\, z') \mid z \,\right].
  \]
  If we use $v$ to \textbf{look ahead}, we can get a \textbf{different function} $\mathcal{T}v$,
  \[
    (\mathcal{T} v)(b, z) \;:=\; \max_{c \in [0, b]} \; u(c) + \beta \, \E\!\left[\, v(R(b-c) + z',\, z') \,\big|\, z \,\right].
  \]
  $v$ kind of represents \textbf{beliefs} about the value of being in various states $(b, z)$.

  $\mathcal{T}v$ represents the \textbf{updated beliefs} if you reason by looking ahead one period.
\end{frame}

\begin{frame}{The Bellman Operator}
  \begin{itemize}
    \item Think of $\mathcal{T}$ as an \textbf{operator} that maps a \textbf{function} to a \textbf{function}: $v \mapsto \mathcal{T}v$.
    \item This is the \textbf{Bellman Operator}.
    \item The Bellman equation can be rewritten as,
    \[
      \boxed{\; V \;=\; \mathcal{T} V. \;}
    \]
    \item We can solve the household's problem by finding a \textbf{fixed point} of $\mathcal{T}$.
  \end{itemize}
\end{frame}

\begin{frame}{Existence and Uniqueness}
  We will not prove this, but it is useful to note that Banach's fixed point theorem implies that, under mild conditions, there is a \textbf{unique} fixed point $V$ of $\mathcal{T}$.

  The idea is that $\mathcal{T}$ is a contraction:
  \[
    \norm{\mathcal{T} v_1 - \mathcal{T} v_2}_\infty
      \le \beta \norm{v_1 - v_2}_\infty.
  \]
  Banach implies:
  \begin{enumerate}
    \item A unique fixed point $V$ exists.
    \item Iterating $v_{k+1} = \mathcal{T} v_k$ from any starting $v_0$ converges to $V$ at rate $\beta$.
  \end{enumerate}
\end{frame}

\begin{frame}{Once More: The Bellman Equation}
\[
  \boxed{\;
    V(\text{state}) = \max_{\text{action}}\ \text{payoff}(\text{state, action}) + \beta \, \E\!\left[\, V(\,\text{next state}\,) \,\big|\, \text{state, action} \,\right]
  \;}
\]
The rest of this course will basically be variations on this.
\end{frame}
