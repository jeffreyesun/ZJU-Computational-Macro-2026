\section{Setup}
% =============================================================

\begin{frame}{Bellman Equation}
  \[
    \boxed{\;
      V(b, z) \;=\; \max_{c \in [0, b]} \; u(c)
        + \beta \, V(R(b-c) + z,\, z)\quad\forall b,z
    \;}
  \]
  This equation characterizes the function $V$ that is needed to solve the household's problem.
\end{frame}

\begin{frame}{$V$ is Unique and Computable}
  For any bounded $v_1, v_2$,
  \[
    \norm{\mathcal{T} v_1 - \mathcal{T} v_2}_\infty
      \;\le\; \beta \, \norm{v_1 - v_2}_\infty.
  \]
  Informally, the Banach fixed-point theorem implies:
  \begin{itemize}
    \item A unique $V^*$ exists that satisfies $V^* = \mathcal{T} V^*$.
    \item Value function iteration will always find $V^*$.
  \end{itemize}
  This lecture: replace the continuous state $b$ with a finite grid, code up $\mathcal{T}$ as a Julia function, iterate to the fixed point, plot $V$.
\end{frame}

\section{Discretization}
% =============================================================

\begin{frame}{Discrete Wealth Grid}
  \begin{itemize}
    \item $V$ is a function on $[0, \infty) \times W$. We can't represent it exactly.
    \item Pick $N$ wealth values $b_1 < b_2 < \cdots < b_N$.
    \item Represent $V$ as the vector $[V(b_1), \ldots, V(b_N)]$.
    \item This makes the function finite-dimensional.
  \end{itemize}
  \textbf{Note:} None of the concepts of value function iteration \textbf{depend} on $V$ being represented as a vector. $V$ could be a spline or a neural network or over a dynamic grid and the concepts would still be valid.
\end{frame}

\begin{frame}[fragile]{Linear vs Log Spacing}
  \begin{lstlisting}
  loggrid(lo, hi, N) = exp.(range(log(lo), log(hi); length=N))

  # linear: equally spaced
  b_grid = collect(range(0.1, 20.0; length=200))

  # log: denser near the borrowing constraint
  b_grid = loggrid(0.1, 20.0, 200)
  \end{lstlisting}
  The policy function is more nonlinear for small $b$, so we use log spacing to put more grid points on smaller $b$.
\end{frame}

\begin{frame}[fragile]{Utility}
  \begin{lstlisting}
  u(c) = c <= 0 ? -Inf : log(c)
  \end{lstlisting}
  Generalized version is CRRA with $\gamma$, but keep it simple for now.
\end{frame}

\begin{frame}[fragile]{Bellman Operator $\mathcal{T}$ as a Julia Function}
  \begin{lstlisting}
  function bellman_operator(V, params)
      (; β, R, z, b_grid) = params

      # end-of-period wealth needed to reach each b' on the grid
      b_end = (b_grid' .- z) ./ R

      # u(c) + β V(b') for every (b, b'); take max over b'
      return maximum(u.(b_grid .- b_end) .+ β.*V'; dims=2)
  end
  \end{lstlisting}
  \texttt{b\_grid} is a column vector. \texttt{b\_grid'} is a row vector.
  
  Each row corresponds to a $b$ value; each column to a $b'$ value. The last line takes \texttt{max} \textbf{within} each $b$ (rows), \textbf{across} $b'$ (columns).
\end{frame}

\section{Value Function Iteration}
% =============================================================

\begin{frame}{Value Function Iteration}
  \begin{enumerate}
    \item Start with $V_0 \equiv 0$ (any initial guess works).
    \vspace{0.4em}
    \item $V_{k+1} \leftarrow \mathcal{T} V_k$.
    \vspace{0.4em}
    \item Stop when $\norm{V_{k+1} - V_k}_\infty < \text{tol}$.
  \end{enumerate}
\end{frame}

\begin{frame}[fragile]{\texttt{solve\_vfi}}
  \begin{lstlisting}
  function solve_vfi(params; tol=1e-6, maxiter=2000, verbosity=0)
      V = zeros(size(params.b_grid))

      iters, Δ = 0, Inf
      while Δ >= tol
          TV = bellman_operator(V, params)
          V, Δ = TV, maximum(abs.(TV .- V))
          iters += 1
          iters > maxiter && error("VFI did not converge in $maxiter iterations")
      end

      verbosity >= 1 && println("VFI converged in $iters iterations with error ", Δ)
      return V
  end
  \end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Baseline Calibration}
  \begin{lstlisting}
  b_grid = loggrid(0.05, 40.0, 400)
  params = (; β=0.96, R=1.04, z=1.0, b_grid)

  V = solve_vfi(params; verbosity=1)
  \end{lstlisting}
  \begin{itemize}
    \item \small $\beta R < 1$, otherwise $V^*$ is infinite.
    \item Grid: 400 log-spaced points from $0.05$ to $40$.
    \item \texttt{params} is a \texttt{NamedTuple} (named tuple). The last \texttt{b\_grid} is equivalent to \texttt{b\_grid=b\_grid}.
  \end{itemize}
\end{frame}

\begin{frame}{Features of Value Function}
  The shape of $V$ is similar to the shape of $u$\footnote{Sheeran (2017)},
  \begin{itemize}
    \item $V(b)$ is increasing (more wealth $\Rightarrow$ more future consumption).
    \item $V(b)$ is concave: an additional unit of wealth is worth more to the poor than to the rich.
    \item The slope is the \textbf{marginal value of wealth}: $V'(b) = u'(c^\star(b))$ by the envelope theorem.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Policy Function $c^\star$}
  \[
    c^\star(b) \;=\; \arg\max_{c \in [0, b]} \; u(c) + \beta \, V\!\big(R(b - c) + z\big).
  \]
  \begin{lstlisting}
  function policy_function(V, params)
      (; β, R, z, b_grid) = params

      # same setup as bellman_operator
      b_end = (b_grid' .- z) ./ R

      # argmax over b' (columns) — return the chosen b'-index for each row
      idx = argmax(u.(b_grid .- b_end) .+ β.*V'; dims=2)
      return vec(getindex.(idx, 2))::Vector{Int}
  end
  \end{lstlisting}
  Same as \texttt{bellman\_operator}, except with \texttt{argmax} instead of \texttt{maximum}.
  
  Returns a vector of integers: grid index of the optimal $b'$ for each (row) $b$.
\end{frame}
