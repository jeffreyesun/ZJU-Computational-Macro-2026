\begin{frame}{Bellman Equation \\ 贝尔曼方程}
\[
  \boxed{\;
    V(b, w) \;=\; \max_{c \in [0, b]} \; u(c)
      + \beta \, \E\!\left[\, V(R(b-c) + w',\, w') \,\big|\, w \,\right]\quad\forall b,w
  \;}
\]
This equation characterizes the function $V$ that is needed to solve the household's problem. / 此方程刻画了求解家庭问题所需的函数 $V$。

\end{frame}

\begin{frame}{$V$ is Unique and Computable \\ $V$ 唯一且可计算}
\begin{columns}[T]
\begin{column}{0.46\textwidth}
For any bounded $v_1, v_2$,
\end{column}
\hfill
\begin{column}{0.46\textwidth}
对任意有界的 $v_1, v_2$：
\end{column}
\end{columns}
\[
  \norm{\mathcal{T} v_1 - \mathcal{T} v_2}_\infty
    \;\le\; \beta \, \norm{v_1 - v_2}_\infty.
\]

\begin{columns}[T]
\begin{column}{0.46\textwidth}
  Informally, the Banach fixed-point theorem implies:
  \begin{itemize}
    \item A unique $V^*$ exists that satisfies $V^* = \mathcal{T} V^*$.
    \item Value function iteration will always find $V^*$.
  \end{itemize}
\end{column}
\hfill
\begin{column}{0.46\textwidth}
  非正式地，Banach 不动点定理意味着：
  \begin{itemize}
    \item 存在唯一的 $V^*$ 满足 $V^* = \mathcal{T} V^*$。
    \item 价值函数迭代总能找到 $V^*$。
  \end{itemize}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Today \\ 今天}
\begin{columns}[T]
\begin{column}{0.46\textwidth}
\begin{enumerate}
  \item Replace the continuous state $b$ with a finite grid.
  \vspace{0.4em}
  \item Code $\mathcal{T}$ as a Julia function.
  \vspace{0.4em}
  \item Iterate $v_{k+1} = \mathcal{T} v_k$ until $\norm{v_{k+1} - v_k}_\infty < \text{tol}$.
  \vspace{0.4em}
  \item Plot $V$.
\end{enumerate}
\end{column}
\hfill
\begin{column}{0.46\textwidth}
\begin{enumerate}
  \item 把连续状态 $b$ 换成有限网格。
  \vspace{0.4em}
  \item 用 Julia 写出 $\mathcal{T}$。
  \vspace{0.4em}
  \item 迭代 $v_{k+1} = \mathcal{T} v_k$，直到 $\norm{v_{k+1} - v_k}_\infty < \text{tol}$。
  \vspace{0.4em}
  \item 画出 $V$。
\end{enumerate}
\end{column}
\end{columns}
\end{frame}

\begin{frame}[fragile]{Bellman Operator $\mathcal{T}$ as a Julia Function \\ 贝尔曼算子 $\mathcal{T}$ 的 Julia 实现}
\begin{lstlisting}
function bellman_operator(V, params)
    (; β, R, y, b_grid) = params

    # end-of-period wealth needed to reach each b' on the grid
    b_end = (b_grid' .- y) ./ R

    # u(c) + β V(b') for every (b, b'); take max over b'
    return maximum(u.(b_grid .- b_end) .+ β .* V'; dims = 2)
end
\end{lstlisting}

\footnotesize
\begin{columns}[T]
\begin{column}{0.46\textwidth}
  \texttt{b\_grid} is a column vector. \texttt{b\_grid'} is a row vector. Each row corresponds to a $b$ value; each column to a $b'$ value. The last line takes \texttt{max} \textbf{within} each $b$ (rows), \textbf{across} $b'$ (columns).
\end{column}
\hfill
\begin{column}{0.46\textwidth}
  \texttt{b\_grid} 是列向量，\texttt{b\_grid'} 是行向量。每一行对应一个 $b$ 值，每一列对应一个 $b'$ 值。最后一行在每个 $b$ 内\textbf{（行内）}、跨所有 $b'$ \textbf{（列间）}取最大。
\end{column}
\end{columns}
\end{frame}

\section{Value Function Iteration / 价值函数迭代}
% =============================================================

\begin{frame}{Value Function Iteration \\ 价值函数迭代}
\begin{columns}[T]
\begin{column}{0.46\textwidth}
\begin{enumerate}
  \item Start with $V_0 \equiv 0$ (any initial guess works).
  \vspace{0.4em}
  \item $V_{k+1} \leftarrow \mathcal{T} V_k$.
  \vspace{0.4em}
  \item Stop when $\norm{V_{k+1} - V_k}_\infty < \text{tol}$.
\end{enumerate}

\vspace{0.6em}

Banach theorem $\Rightarrow$ algorithm converges to true $V^*$ at rate $\beta$.
\end{column}
\hfill
\begin{column}{0.46\textwidth}
\begin{enumerate}
  \item 从 $V_0 \equiv 0$ 开始（任意初值都行）。
  \vspace{0.4em}
  \item $V_{k+1} \leftarrow \mathcal{T} V_k$。
  \vspace{0.4em}
  \item 当 $\norm{V_{k+1} - V_k}_\infty < \text{tol}$ 时停。
\end{enumerate}

\vspace{0.6em}

由 Banach 不动点定理 $\Rightarrow$ 算法以速率 $\beta$ 收敛到真实的 $V^*$。
\end{column}
\end{columns}
\end{frame}

\begin{frame}[fragile]{\texttt{solve\_vfi} / 求解函数}
\begin{lstlisting}
function solve_vfi(params; tol = 1e-6, maxiter = 2000, verbosity = 0)
    V = zeros(size(params.b_grid))

    Δ, iters = Inf, 0
    while Δ >= tol
        TV = bellman_operator(V, params)
        Δ  = maximum(abs.(TV .- V))
        V  = TV
        iters += 1
        iters > maxiter && error("VFI did not converge in $maxiter iterations")
    end

    verbosity >= 1 && println("VFI converged in $iters iterations with error $Δ")
    return V
end
\end{lstlisting}
\end{frame}

\vspace{0.3em}

\texttt{params} 是一个 \texttt{NamedTuple}（具名元组）。末尾的 \texttt{b\_grid} 等价于 \texttt{b\_grid = b\_grid}。
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Features of Value Function \\ 价值函数的性质}
\begin{columns}[T]
\begin{column}{0.46\textwidth}
$V(b)$ is increasing because more wealth never hurts.

\vspace{0.4em}

$V(b)$ is concave: an additional unit of wealth is worth more to the poor than to the rich.

\vspace{0.4em}

The slope is the \textbf{marginal value of wealth}: $V'(b) = u'(c^\star(b))$ by the envelope theorem.
\end{column}
\hfill
\begin{column}{0.46\textwidth}
$V(b)$ 递增：财富越多绝不会更糟。

\vspace{0.4em}

$V(b)$ 凹：多一单位财富对穷人比对富人更有价值。

\vspace{0.4em}

斜率即\textbf{财富的边际价值}：由包络定理，$V'(b) = u'(c^\star(b))$。
\end{column}
\end{columns}
\end{frame}
