\section{阶段与外循环}
% =============================================================

\begin{frame}{阶段（Stages）}
  \begin{itemize}
    \item 异质性主体（heterogeneous-agent）宏观模型由\textbf{阶段}（Stages）与\textbf{外循环}（Outer Loops）组合而成。
    \item \textbf{阶段}是基本构件。它们定义家庭问题，而家庭问题正是\textbf{最难的部分}。
    \item 由于\textbf{阶段}在概念上\textbf{定义良好}、\textbf{模块化}且\textbf{可组合}，复杂的宏观模型可以通过简单地组合\textbf{阶段}来构建。
  \end{itemize}
\end{frame}

\begin{frame}{阶段求解家庭模块}
  \begin{itemize}
    \item 一个\textbf{阶段}定义了家庭在给定\textbf{环境}（即\textbf{价格}与\textbf{参数}）下如何做决策。
    \item \textbf{阶段}组合成\textbf{家庭模块}（household block）：给定价格与参数，给出值函数与人口分布。
    \item 这就给出了\textbf{总量}（aggregates）：商品需求、资本需求、劳动供给，\textbf{作为}价格、参数、市场紧度等的函数。
    \item 在某种意义上，可以把家庭模块看作经济的\textbf{一侧}，有点像一条大的总需求曲线。
  \end{itemize}
\end{frame}

\begin{frame}{为什么用阶段？}
  \begin{itemize}
    \item 阶段是\textbf{可组合的}（composable）：复杂模型由若干阶段组合而成。
    \item 阶段具有\textbf{精确的数学定义}：模型可以被定义并求解，而无需操心实现细节。
    \item 阶段是\textbf{标准化的}（standardized）：存在可下载并复用的优化实现（例如 \texttt{HouseholdStages.jl}、SSJ Toolkit 或 ECON-ARK）。
  \end{itemize}

\end{frame}

\begin{frame}{家庭模块如何嵌入其中}
  \textbf{家庭模块}（由\textbf{阶段}定义并求解）只是模型的一部分。

  可以把模型的其余部分视为由以下几部分组成：
  \begin{itemize}
    \item 一组描述经济其余部分（厂商、政府等）的方程，通常实现为函数的\textbf{有向无环图}（directed acyclic graph, DAG）。
    \item 一系列\textbf{外循环}（Outer Loops），用于求解均衡价格、稳态、转移路径，或带有总量不确定性的全局解。
  \end{itemize}
\end{frame}

\begin{frame}{外循环：举例}
  均衡：在价格上搜索，使\textbf{市场出清}（market clearing）或\textbf{超额需求}（excess demand）目标为零。

  稳态（steady state）：在人口分布 $\Lambda$ 上搜索，使\textbf{人口演化}目标 $\Lambda^\text{end} - \Lambda^{\text{start}'}$ 为零。

  校准（calibration）：在参数上搜索，使\textbf{模型拟合}目标 $\text{moment}^\text{model} - \text{moment}^\text{data}$ 为零。

  完美预期转移（perfect foresight transition）：在市场出清价格的\textbf{路径}上搜索，使``路径形式的超额需求''目标为零。

  总量不确定性（aggregate uncertainty）：在\textbf{全局值函数}上搜索，使\textbf{外层贝尔曼目标}为零。
\end{frame}

\begin{frame}{Aiyagari 模型}
  Aiyagari 模型由以下三个模块组成：
  \begin{enumerate}
    \item 由\textbf{阶段}构成的\textbf{家庭模块}。
    \item 把资本与劳动转化为商品的\textbf{厂商}（firm）。
    \item 一种\textbf{均衡}（equilibrium，即稳态）概念。
  \end{enumerate}

  熟悉的经济学：
  \begin{enumerate}
    \item 一个需求侧。
    \item 一个供给侧。
    \item 一种均衡概念。
  \end{enumerate}
\end{frame}

\section{Aiyagari 家庭模块}
% =============================================================

\begin{frame}{与上次相同}

  上次讲过的三阶段家庭模块：
  \begin{enumerate}
    \item 收入冲击（$z$ 上的马尔可夫链（Markov chain））。
    \item 收入（手头现金 $b = R\,b^{\mathrm{end}} + z$）。
    \item 消费--储蓄（在 $b^{\mathrm{end}}$ 上做网格搜索）。
  \end{enumerate}

  给定价格的稳态解：
  \begin{itemize}
    \item VFI：值函数 $V$ 与策略 $c^\star$。
    \item 向前模拟：平稳分布 $\lambda^\star$。
    \item 今天：将 $R$ 视为需要求解的内生\emph{价格}。
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{在给定 $R$ 下求解家庭模块}
  \begin{lstlisting}
  function solve_aiyagari_steady_state(env; verbosity=1)
      V_end = solve_vfi(env; verbosity)
      return (; V_end, λ = find_steady_state_population(V_end, env; verbosity))
  end
  \end{lstlisting}

  \begin{itemize}
    \item env 包含价格与参数。
    \item 一次函数调用，就能求出稳态。
    \item 可以把它（除其它解释外）理解为一个大函数 $R \mapsto \bar K$。
  \end{itemize}
\end{frame}

\section{总量}
% =============================================================

\begin{frame}{总量矩作为内积}
  每一个总量矩都是一个\textbf{对偶向量}（dual vector）：与 $\lambda^\star$ 的内积。
  \begin{align*}
    \bar K &= \sum_{i,j} b_i \, \lambda^\star_{ij} \\
    \bar L &= \sum_{i,j} \lambda^\star_{ij}        \\
    \bar C &= \sum_{i,j} c^\star_{ij} \, \lambda^\star_{ij} \\
    \bar V &= \sum_{i,j} V_{ij} \, \lambda^\star_{ij}
  \end{align*}
\end{frame}

\begin{frame}[fragile]{\texttt{compute\_household\_aggregates}}
  \begin{lstlisting}
  function compute_household_aggregates(;param_vals...)
      params = get_params(;param_vals...)
      (V_end, λ) = solve_aiyagari_steady_state(params)

      b_grid_policy = policy_function(V_end, params)
      c_policy = get_c_policy(params.b_grid, b_grid_policy)

      L_agg = sum( λ)
      K_agg = sum(params.b_grid .* λ)
      V_agg = sum(V_end .* λ)
      C_agg = sum(c_policy .* λ)

      return (; V_end, λ, L_agg, K_agg, V_agg, C_agg)
  end
  \end{lstlisting}
\end{frame}

\section{厂商}
% =============================================================

\begin{frame}[fragile]{Cobb--Douglas}
  代表性厂商，Cobb--Douglas 生产函数：
  \[
    Y = K^\alpha L^{1-\alpha}.
  \]
  这是模型的供给侧。

  代码中，
  \begin{lstlisting}
  compute_Y(K, L; α = 1/3) = K^ α * L^ (1 - α)
  \end{lstlisting}
\end{frame}

\section{均衡}
% =============================================================

\begin{frame}{商品市场出清}
  在稳态下，
  \[
    \bar C \;=\; Y.
  \]
  定义商品的\emph{超额需求}：
  \[
    \text{excess\_demand}(R) = \frac{\bar C(R) - Y(R)}{Y(R)}.
  \]
  均衡 $R$ 即该函数的\textbf{根}：使 $\text{excess\_demand}(R) = 0$ 成立的取值。
\end{frame}

\begin{frame}[fragile]{\texttt{compute\_excess\_demand}}
  我们可以把这个函数实现出来，
  \begin{lstlisting}
  function compute_excess_demand(R)
      res = compute_household_aggregates(;R)
      Y_agg = compute_Y(res.K_agg, res.L_agg)
      return excess_demand = (res.C_agg - Y_agg)/Y_agg
  end
  \end{lstlisting}
  直接影响的链条为：
  \[
    R \to V \to c^\star \to \lambda^\star \to (\bar K, \bar C) \to Y.
  \]
\end{frame}

\begin{frame}[fragile]{求解均衡 $R$}
  \begin{lstlisting}
  tol = 0.01
  lr = 0.001

  R = 1.01
  err = Inf
  while err > tol
      excess_demand = compute_excess_demand(R)

      R_new = R - excess_demand*lr
      err = abs(excess_demand)

      R = R_new

      println("R=$R\t excess_demand=$excess_demand")
  end
  \end{lstlisting}

  \begin{itemize}
    \item Tâtonnement（试探调价）：一种简单的梯度式更新——沿商品超额需求的方向更新 $R$。
    \item 若 $\bar C > Y$，则家庭消费过多——$\uparrow R\Rightarrow \uparrow$ 储蓄 $\Rightarrow \uparrow K, Y$。
    \item 若 $\bar C < Y$，则家庭储蓄过多——$\downarrow R\Rightarrow \downarrow$ 储蓄 $\Rightarrow \downarrow K, Y$。
  \end{itemize}
\end{frame}
