\begin{frame}{模型结构}
  本系列幻灯片：使用 Julia 求解带或不带总量冲击的异质性主体（heterogeneous-agent, HA）一般均衡（general equilibrium, GE）模型。

  我们涵盖具有以下形式的离散时间 GE 模型：
  \begin{enumerate}
    \item 由若干\textbf{阶段}（Stages）构成的前瞻性家庭模块
    \item 一组描述经济其余部分（厂商、政府等）的方程，通常表示为函数的\textbf{有向无环图}（directed acyclic graph, DAG）
    \item 一组\textbf{外循环}（outer loops），用于求解均衡、稳态、校准、转移路径或全局解等
  \end{enumerate}
\end{frame}

\begin{frame}{课程结构}
  \begin{itemize}
    \item 第 1--4 讲：阶段（Stages）。
    \item 第 5--8 讲：外循环（outer loops）。
    \begin{itemize}
      \item 最终引出``神经价值函数迭代''（neural Value Function Iteration, nVFI）全局求解算法
    \end{itemize}
  \end{itemize}
\end{frame}

\section{动机}
% =============================================================

\begin{frame}{异质性主体：动机一}
  \begin{itemize}
    \item 经验上的平均边际消费倾向（marginal propensity to consume, MPC）较高，但平均储蓄\textbf{同样}较高。
    \item 典型的代表性主体模型难以解释这一点。
    \item 然而，如果穷人具有高 MPC 与低储蓄，富人具有低 MPC 与高储蓄，那么\textbf{异质性主体}模型就能解释。
  \end{itemize}
\end{frame}

\begin{frame}{异质性主体：动机二}
\begin{itemize}
  \item 政策（例如利率政策）的变化会以\textbf{不同方式}影响不同的人。
  \item 一项政策的\textbf{总量}效应取决于它针对的是谁。
  \item 我们关心政策的分布效应与总量后果。
\end{itemize}

\end{frame}

\section{家庭问题：两期模型}
% =============================================================

\begin{frame}{微观基础：单个家庭}
  \begin{itemize}
    \item 宏观经济学研究的是个体行为如何塑造总量结果。
    \item 要理解总量结果的来源，必须研究个体行为的来源。
  \end{itemize}
\end{frame}

\begin{frame}{两期模型}
  家庭在第 0 期持有\textbf{财富} $b_0$。第 1 期获得收入 $z_1$。
  \[
    \max_{c_0, c_1 \ge 0} \; u(c_0) + \beta u(c_1)
  \]
  \[
    \text{s.t.}\quad b_0 + \frac{z_1}{R} \geq c_0 + \frac{c_1}{R}.
  \]
  \begin{itemize}
    \item 如果今天储蓄 \$1，明天可以消费 \$$R$，但效用按 $\beta$ 折现。
    \item 拉格朗日函数：$\mathcal{L} = u(c_0) + \beta u(c_1) - \lambda\p{c_0 + \frac{c_1}{R} - b_0 - \frac{z_1}{R}}$。
  \end{itemize}
\end{frame}

\begin{frame}{欧拉方程}
  拉格朗日 FOC：
  \begin{align*}
    u'(c_0) &= \lambda, \\
    \beta u'(c_1) &= \frac{\lambda}{R}.
  \end{align*}
  整理得，
  \[
    \boxed{\frac{u'(c_0)}{u'(c_1)} = \beta R.}
  \]
  在最优点，家庭对于``今天多消费一单位''与``今天储蓄一单位、明天获得 $R$ 单位''是\textbf{无差异}的。
\end{frame}

\begin{frame}{例：对数效用下的解析解}
  设 $u(c) = \log c$。则 $c_1 = \beta R c_0$，且
  \[
    c_0^\star = \frac{b_0 + \frac{z_1}{R}}{1+\beta},
    \qquad
    c_1^\star = \frac{\beta R \, b_0 + \frac{z_1}{R}}{1+\beta}.
  \]
  取 $\beta = 0.96$、$R = 1.04$、$b_0 = 10$、$z_1 = 2$：
  \[
    c_0^\star \approx 6.08, \;\; c_1^\star \approx 6.07.
  \]
  存在\textbf{消费平滑}（consumption smoothing）。收入序列 $(10, 2)$ 起伏较大，消费序列 $(6.08, 6.07)$ 则平稳得多。
\end{frame}

\begin{frame}{策略函数}
  \textbf{策略函数}（policy function）以\textbf{初始财富} $b_0$ 表示第 0 期的最优消费：
  \[
    c_0^\star(b_0) = \frac{b_0 + z_1/R}{1+\beta}.
  \]
  这样我们就建立了一个关于家庭行为以及\textbf{家庭行为}如何产生的模型。
\end{frame}

\begin{frame}{CRRA 偏好}
  常用的假设是常相对风险厌恶（constant relative risk aversion, CRRA）效用：
  \[
    u(c) = \begin{cases}
      \frac{c^{1-\gamma}-1}{1-\gamma}, & \gamma > 0, \gamma \neq 1, \\
      \log c, & \gamma = 1.
    \end{cases}
  \]
  效用关于 $c$ \textbf{递增}：
  \[
    \frac{\partial u(c)}{\partial c} = c^{-\gamma} > 0.
  \]
  效用为\textbf{凹函数}：边际效用关于 $c$ \textbf{递减}：
  \[
    \frac{\partial^2 u(c)}{\partial c^2} = -\gamma c^{-\gamma-1} < 0.
  \]
\end{frame}

\section{家庭问题：$T$ 期模型}

\begin{frame}{$T$ 期模型}
  终生效用为
  \[
    \max_{c_0, \ldots, c_{T-1}} \sum_{t=0}^{T-1} \beta^t u(c_t),
    \quad b_{t+1} = R(b_t - c_t) + z_{t+1}.
  \]
  求解拉格朗日函数会得到 $T-1$ 个欧拉方程：
  \[
    u'(c_t) = \beta R \, u'(c_{t+1}).
  \]
  如果未来收入完全可预见，问题仍然容易求解。

  那如果收入是随机的呢？
\end{frame}

\begin{frame}{随机收入}
  \begin{itemize}
    \item 假设 $\{z_t\}$ 服从转移矩阵为 $\Pi$ 的马尔可夫链（Markov chain）。
    \item 给定今天的 $z_t$，我们知道 $z_{t+1}$ 的\emph{条件分布}，但不知道实现值。
    \item 在每一期 $t$，家庭在不知道 $z_{t+1}$ 的情况下选择 $c_t$。
  \end{itemize}
\end{frame}

\begin{frame}{随机欧拉方程}
  取 $T = 2$，可代入约束：
  \[
    c_1 = R(b_0 - c_0) + z_1.
  \]
  问题变为
  \[
    \max_{c_0} u(c_0) + \beta \E\left[\, u\left(R(b_0 - c_0) + z_1\right) \right],
  \]
  对 $c_0$ 的一阶条件为
  \[
    \boxed{\;u'(c_0) = \beta R \, \E\left[ u'(c_1) \right].}
  \]
  欧拉方程现在包含了对\emph{不确定的} $c_1$ 的期望。
\end{frame}

\begin{frame}{无限期问题（$T = \infty$）}
  令人意外的是，当 $T\to\infty$ 时问题反而稍微容易些：
  \begin{align*}
    \max \E_0 & \ \sum_{t=0}^{\infty} \beta^t u(c_t)\\
    \text{s.t.}\quad b_{t+1} &= R(b_t - c_t) + z_{t+1}.
  \end{align*}
  但我们究竟在对什么取 $\max$？严格地说，需要找到一整个\textbf{策略函数}序列：
  \[
    c_t = c_t(z_0, \ldots, z_t) \quad \forall t,
  \]
  这看上去几乎不可行。所幸我们可以通过重新表述来简化问题。

  （注：此处略去横截条件（transversality condition）的问题。）
\end{frame}

\section{递归形式}
% =============================================================

\begin{frame}{Richard Bellman 的第一个观察}
  在每一时刻 $t$，家庭只做一个选择，然后等待：

  \[
    \max_{c_t} = u(c_t) + \beta \E_t\left[ \max \E_{t+1}\sum_{s=t+1}^\infty \beta^t u(c_s) \right]
  \]
  \[
    s.t. \quad b_{s+1} = R(b_s - c_s) + z_{s+1} \quad\forall s \ge t.
  \]

  关键在于，$t$ 时的问题与 $t+1$ 时的问题\textbf{几乎}完全相同。

  仅有的差异可能来自 $b_t$ 与 $z_t$。它们就是\textbf{状态变量}（state variables）。
\end{frame}

\begin{frame}{状态变量}
  \begin{itemize}
    \item 非正式地说，\textbf{状态变量}是那些让不同时期（以及不同决策）产生实质性差别的变量。
    \item 在这里，状态变量是财富 $b_t$ 与收入 $z_t$。给定它们，家庭的未来条件期望效用与最优决策就被唯一决定。
  \end{itemize}
\end{frame}

\begin{frame}{Bellman 的第二与第三个观察}
  Bellman 还意识到另外两点：
  \begin{enumerate}
    \item[2.] 可以将第一个洞见表述为关于\textbf{值函数}（value function）$V$ 的方程。
    \item[3.] 一旦知道这个\textbf{值函数}，就能计算\textbf{策略函数}并（数值地）求解整个模型。
  \end{enumerate}
  用计算机科学的话说，这是一种\textbf{缓存}（caching）。
\end{frame}

\begin{frame}{值函数与贝尔曼方程}
  定义\emph{值函数} $V(b, z)$ 为状态为 $(b, z)$ 时未来效用的\textbf{净现值}：
  \begin{align*}
    V(b, z) &= \max_{c\in[0,b]} \mathbb{E}_0 \sum_{s=0}^\infty \beta^{s} u(c_s)\\
    \text{s.t.} \quad b_{s+1} &= R(b_s - c_s) + z_{s+1}\ \forall s.
  \end{align*}
  $\quad$（前提是未来所有选择也都是最优的。）

  那么 $V$ 满足\textbf{贝尔曼方程}（Bellman equation）：
  \[
    \boxed{V(b, z) = \max_{c \in [0, b]} \; u(c) + \beta \, \E\!\left[\, V(b', z') \,\big|\, z \,\right]}
  \]
  s.t. $b' = R(b-c) + z'$
\end{frame}

\begin{frame}{值函数就是一切}
  一旦知道 $V$，就可以计算\textbf{策略函数}并求解整个（家庭）模型：
  \[
  c^\star(b, z) = \arg\max_{c \in [0, b]} u(c) + \beta \, \E\!\left[\, V(R(b-c) + z,\, z') \mid z \,\right].
  \]
\end{frame}

\begin{frame}{如何找到 $V$？}
  \begin{itemize}
    \item 但仍然存在一个问题：我们需要找到的不是一个数，而是一整个\textbf{函数} $V$。
    \item 可以通过``值函数迭代''（Value Function Iteration, VFI）做到这一点。
    \item 为理解 VFI：先设想我们猜测了一个值函数，并且猜\textbf{错}了。
  \end{itemize}

\end{frame}

\begin{frame}{贝尔曼算子作用于任意 $v$}
  假设我们猜测了一个\textbf{函数} $v$。即便 $v$ 可能是\textbf{错的}，它仍然会预测某个策略：
  \[
    c^*(b, z; v) = \arg\max_{c \in [0, b]} u(c) + \beta \, \E\!\left[\, v(R(b-c) + z',\, z') \mid z \,\right].
  \]
  若用 $v$ 进行\textbf{前瞻}，可以得到\textbf{另一个函数} $\mathcal{T}v$：
  \[
    (\mathcal{T} v)(b, z) \;:=\; \max_{c \in [0, b]} \; u(c) + \beta \, \E\!\left[\, v(R(b-c) + z',\, z') \,\big|\, z \,\right].
  \]
  $v$ 可以被理解为对处在状态 $(b, z)$ 的价值的\textbf{信念}。

  $\mathcal{T}v$ 则表示通过向前看一期之后得到的\textbf{更新后的信念}。
\end{frame}

\begin{frame}{贝尔曼算子}
  \begin{itemize}
    \item 可以把 $\mathcal{T}$ 看作把\textbf{函数}映射到\textbf{函数}的\textbf{算子}：$v \mapsto \mathcal{T}v$。
    \item 这就是\textbf{贝尔曼算子}（Bellman operator）。
    \item 贝尔曼方程可重写为
    \[
      \boxed{\; V \;=\; \mathcal{T} V. \;}
    \]
    \item 求解家庭问题相当于寻找 $\mathcal{T}$ 的\textbf{不动点}（fixed point）。
  \end{itemize}
\end{frame}

\begin{frame}{存在性与唯一性}
  我们不打算证明，但值得指出：Banach 不动点定理（Banach's fixed point theorem）保证，在温和条件下，$\mathcal{T}$ 存在\textbf{唯一}的不动点 $V$。

  关键在于 $\mathcal{T}$ 是一个压缩映射（contraction）：
  \[
    \norm{\mathcal{T} v_1 - \mathcal{T} v_2}_\infty
      \le \beta \norm{v_1 - v_2}_\infty.
  \]
  Banach 定理由此给出：
  \begin{enumerate}
    \item 存在唯一的不动点 $V$。
    \item 从任意初始 $v_0$ 出发，迭代 $v_{k+1} = \mathcal{T} v_k$ 以速率 $\beta$ 收敛到 $V$。
  \end{enumerate}
\end{frame}

\begin{frame}{再看一次：贝尔曼方程}
\[
  \boxed{\;
    V(\text{状态}) = \max_{\text{动作}}\ \text{payoff}(\text{状态, 动作}) + \beta \, \E\!\left[\, V(\,\text{下一状态}\,) \,\big|\, \text{状态, 动作} \,\right]
  \;}
\]
本课程余下的内容基本上都是这一公式的变种。
\end{frame}
