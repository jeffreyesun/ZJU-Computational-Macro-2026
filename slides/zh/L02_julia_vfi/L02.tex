\section{准备}
% =============================================================

\begin{frame}{贝尔曼方程（Bellman equation）}
  \[
    \boxed{\;
      V(b, z) \;=\; \max_{c \in [0, b]} \; u(c)
        + \beta \, V(R(b-c) + z,\, z)\quad\forall b,z
    \;}
  \]
  此方程刻画了求解家庭问题所需的函数 $V$。
\end{frame}

\begin{frame}{$V$ 的唯一性与可计算性}
  对任意有界的 $v_1, v_2$，
  \[
    \norm{\mathcal{T} v_1 - \mathcal{T} v_2}_\infty
      \;\le\; \beta \, \norm{v_1 - v_2}_\infty.
  \]
  非正式地，Banach 不动点定理（Banach fixed-point theorem）意味着：
  \begin{itemize}
    \item 存在唯一的 $V^*$ 满足 $V^* = \mathcal{T} V^*$。
    \item 值函数迭代（value function iteration, VFI）总能找到 $V^*$。
  \end{itemize}
  本讲：用有限网格代替连续状态 $b$，把 $\mathcal{T}$ 写成一个 Julia 函数，迭代到不动点，并画出 $V$。
\end{frame}

\section{离散化}
% =============================================================

\begin{frame}{离散财富网格}
  \begin{itemize}
    \item $V$ 是定义在 $[0, \infty) \times W$ 上的函数。我们无法精确表示它。
    \item 选取 $N$ 个财富取值 $b_1 < b_2 < \cdots < b_N$。
    \item 把 $V$ 表示为向量 $[V(b_1), \ldots, V(b_N)]$。
    \item 这样函数就变成了有限维的。
  \end{itemize}
  \textbf{注：}值函数迭代的概念\textbf{并不依赖}于 $V$ 是否被表示为向量。$V$ 也可以是样条（spline）、神经网络，或定义在动态网格上的函数，相关概念都同样适用。
\end{frame}

\begin{frame}[fragile]{线性网格 vs 对数网格}
  \begin{lstlisting}
  loggrid(lo, hi, N) = exp.(range(log(lo), log(hi); length=N))

  # linear: equally spaced
  b_grid = collect(range(0.1, 20.0; length=200))

  # log: denser near the borrowing constraint
  b_grid = loggrid(0.1, 20.0, 200)
  \end{lstlisting}
  策略函数（policy function）在 $b$ 较小时非线性更强，因此我们使用对数网格，把更多网格点放在较小的 $b$ 上。
\end{frame}

\begin{frame}[fragile]{效用函数}
  \begin{lstlisting}
  u(c) = c <= 0 ? -Inf : log(c)
  \end{lstlisting}
  更一般的版本是带参数 $\gamma$ 的 CRRA，但目前先保持简单。
\end{frame}

\begin{frame}[fragile]{把贝尔曼算子（Bellman operator）$\mathcal{T}$ 写成 Julia 函数}
  \begin{lstlisting}
  function bellman_operator(V, params)
      (; β, R, z, b_grid) = params

      # end-of-period wealth needed to reach each b' on the grid
      b_end = (b_grid' .- z) ./ R

      # u(c) + β V(b') for every (b, b'); take max over b'
      return maximum(u.(b_grid .- b_end) .+ β.*V'; dims=2)
  end
  \end{lstlisting}
  \texttt{b\_grid} 是列向量，\texttt{b\_grid'} 是行向量。

  每一行对应一个 $b$ 值，每一列对应一个 $b'$ 值。最后一行代码在每个 $b$ \textbf{内}（行内）、\textbf{跨} $b'$（列）取 \texttt{max}。
\end{frame}

\begin{frame}{关于示例代码的说明}
  笔记本中的代码\textbf{并未}做优化，也并不追求速度，只为说明概念。

  更快的实现见 HouseholdStages.jl 软件包。
\end{frame}

\section{值函数迭代}
% =============================================================

\begin{frame}{值函数迭代}
  \begin{enumerate}
    \item 从 $V_0 \equiv 0$ 开始（任何初始猜测均可）。
    \vspace{0.4em}
    \item $V_{k+1} \leftarrow \mathcal{T} V_k$。
    \vspace{0.4em}
    \item 当 $\norm{V_{k+1} - V_k}_\infty < \text{tol}$ 时停止。
  \end{enumerate}
\end{frame}

\begin{frame}[fragile]{\texttt{solve\_vfi}}
  \begin{lstlisting}
  function solve_vfi(params; tol=1e-6, maxiter=2000, verbosity=0)
      V = zeros(size(params.b_grid))

      iters, Δ = 0, Inf
      while Δ >= tol
          TV = bellman_operator(V, params)
          V, Δ = TV, maximum(abs.(TV .- V))
          iters += 1
          iters > maxiter && error("VFI did not converge in $maxiter iterations")
      end

      verbosity >= 1 && println("VFI converged in $iters iterations with error ", Δ)
      return V
  end
  \end{lstlisting}
\end{frame}

\begin{frame}[fragile]{基准校准（Baseline Calibration）}
  \begin{lstlisting}
  b_grid = loggrid(0.05, 40.0, 400)
  params = (; β=0.96, R=1.04, z=1.0, b_grid)

  V = solve_vfi(params; verbosity=1)
  \end{lstlisting}
  \begin{itemize}
    \item \small 需要 $\beta R < 1$，否则 $V^*$ 为无穷大。
    \item 网格：从 $0.05$ 到 $40$ 的 400 个对数间距点。
    \item \texttt{params} 是一个具名元组（\texttt{NamedTuple}）。末尾的 \texttt{b\_grid} 等价于 \texttt{b\_grid=b\_grid}。
  \end{itemize}
\end{frame}

\begin{frame}{值函数的性质}
  $V$ 的形状与 $u$ 的形状相似\footnote{Sheeran (2017)}：
  \begin{itemize}
    \item $V(b)$ 单调递增（更多财富 $\Rightarrow$ 更多未来消费）。
    \item $V(b)$ 是凹函数：对穷人而言，多一单位财富比对富人更有价值。
    \item 斜率即\textbf{财富的边际价值}：由包络定理（envelope theorem），$V'(b) = u'(c^\star(b))$。
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{策略函数 $c^\star$}
  \[
    c^\star(b) \;=\; \arg\max_{c \in [0, b]} \; u(c) + \beta \, V\!\big(R(b - c) + z\big).
  \]
  \begin{lstlisting}
  function policy_function(V, params)
      (; β, R, z, b_grid) = params

      # same setup as bellman_operator
      b_end = (b_grid' .- z) ./ R

      # argmax over b' (columns) — return the chosen b'-index for each row
      idx = argmax(u.(b_grid .- b_end) .+ β.*V'; dims=2)
      return vec(getindex.(idx, 2))::Vector{Int}
  end
  \end{lstlisting}
  与 \texttt{bellman\_operator} 相同，只是把 \texttt{maximum} 换成了 \texttt{argmax}。

  返回一个整数向量：每个（行）$b$ 所对应最优 $b'$ 的网格索引。
\end{frame}
