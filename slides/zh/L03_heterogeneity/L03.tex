\section{策略函数}
% =============================================================

\begin{frame}{策略函数：$c^\star$}
  策略函数（policy function）是\textbf{贝尔曼算子}（Bellman operator）中 \textbf{max} 对应的 \textbf{argmax} 版本：
  \begin{align*}
    \mathcal{T}v(b)        &\;=\; \max_{c \in [0, b]} \; u(c) + \beta \, v\!\big(R(b - c) + z\big) \tag{贝尔曼算子} \\[0.4em]
    c^\star(b)  &\;=\; \arg\max_{c \in [0, b]} \; u(c) + \beta \, v\!\big(R(b - c) + z\big) \tag{策略函数}
  \end{align*}
  \begin{itemize}
    \item 一旦求出真正的 $V$，$\arg\max$ 就给出最优策略 $c^\star$。
    \item 更进一步地，任意的 $v$ 都会给出某个对应的策略函数。
    \begin{itemize}
      \item $v$ 可以理解为关于处于各种状态的价值的\textbf{信念}。
    \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{边际消费倾向（MPC）}
  斜率 $\partial c^\star / \partial b$ 就是\textbf{边际消费倾向}（marginal propensity to consume, MPC）。

  无论在理论上还是经验上，MPC 对较\textbf{穷}的家庭一般较高，对富裕家庭则较低。

  可以将其视为家庭受\textbf{约束}程度的函数。
\end{frame}

\section{符号约定}
% =============================================================

\begin{frame}{符号约定}
  \begin{itemize}
    \item 既然我们已经有了一些直觉，接下来设定一些后面会派上用场的符号约定。
    \item 把状态变量 $z$ 理解为\textbf{人力资本}（human capital），它服从一个\textbf{马尔可夫过程}（Markov process）$\Pi$。
    \item 当期收入为 $zw$，其中 $w$ 是工资水平。目前我们将其标准化为 $w = 1$。
    \item \textbf{状态}现在是 $(b, z)$。
    \item 在计算机上，每个 $V$ 现在都是一个\textbf{矩阵}。行对应财富 $b$，列对应人力资本 $z$。
  \end{itemize}
\end{frame}

\begin{frame}{阶段（Stages）}
  每一项\textbf{模型特征}都可以视为一个\textbf{阶段}（Stage）：
  \begin{enumerate}
    \item 收入冲击（Income Shock）
    \item 收入（Income）
    \item 消费--储蓄（Consumption-Saving）
  \end{enumerate}
  每个\textbf{阶段}都是一个\textbf{简单的子时期}。
  \begin{itemize}
    \item 这一点后面会详细展开，但阶段的\textbf{函数可组合性}（functional composability）类似于连续时间家庭模型的\textbf{加法可组合性}（additive composability）。
    \item 这使我们能够通过组合\textbf{阶段}来构建\textbf{复杂}的模型。
  \end{itemize}
\end{frame}

\section{阶段}

\begin{frame}{阶段 1 --- 收入冲击}
  \begin{itemize}
    \item $\Pi_{ij} = \Pr(z_t = j \mid z^{\mathrm{end}}_{t-1} = i)$，行随机（row-stochastic）。
    \item \textbf{反向迭代}（值函数）。
  \end{itemize}
  \[
    V^{\mathrm{start}} \;=\; V^{\mathrm{pre\_inc}} \, \Pi^\top.
  \]
  $V^{\mathrm{start}}$ 在 $(b^{\mathrm{end}}, z^{\mathrm{end}}_{t-1})$ 处的每一个分量都是在给定 $z^{\mathrm{end}}_{t-1}$ 下对 $z_t$ 取的期望。

  （注：关于为什么矩阵乘法可以表示 $V$ 在马尔可夫链上的反向迭代，可参阅马尔可夫链的标准教材。）
\end{frame}

\begin{frame}{阶段 2 --- 收入}
  这一阶段仅代表收入的\textbf{获得}，
  \[
    b_t = R \, b^{\mathrm{end}}_{t-1} + z_t,
  \]
  本质上是一次变量替换。

  显式地写出反向步骤：
  \begin{align*}
    V^{\mathrm{pre\_inc}}(b^{\mathrm{end}}_{t-1}, z)
    \; &= \; V\!\big(R\, b^{\mathrm{end}}_{t-1} + z,\; z\big).
  \end{align*}

  在示例代码中，我们通过在新的点 $R\, b^{\mathrm{end}}_{t-1} + z$ 上对 $V$ \textbf{重新插值}来实现这一步。
\end{frame}

\begin{frame}{阶段 3 --- 消费--储蓄}
  这一阶段代表 argmax 步骤：
  \[
    b^{\mathrm{end}}_t \;=\; b_t - c^\star(b_t, z_t).
  \]
  显式地写出反向步骤：
  \[
    V(b, z) \;=\; \max_{b^{\mathrm{end}} \in b_{\mathrm{grid}}} u(b - b^{\mathrm{end}}) + \beta V^{\mathrm{end}}(b^{\mathrm{end}}, z).
  \]

  在此我们把以 $\beta$ 表示的时间折现并入消费--储蓄阶段，但一般来说它也可以视为单独的一个阶段。
\end{frame}

\begin{frame}[fragile]{反向迭代}
  \begin{lstlisting}
    function bellman_operator(V_end, params)
      (; β) = params
      # Stage 3
      V_pre_cons = consumption_saving_backward(V_end, params)

      # Stage 2
      V_pre_inc = income_backward(V_pre_cons, params)

      # Stage 1
      V_start = income_shock_backward(V_pre_inc, params)

      # Passage of time
      V_end_new = β .* V_start
      return V_end_new
  end
  \end{lstlisting}
  $V$ 的反向迭代不过是各阶段反向迭代函数的\textbf{函数复合}（functional composition）。
\end{frame}

\section{模拟一个人口}
% =============================================================

\begin{frame}{几乎免费的异质性}
  \begin{itemize}
    \item 之前我们把 $V$ 和 $c^*$ 视为单个家庭在不同\textbf{可能}状态下的价值与策略。
    \item 但也可以把 $V$ 和 $c^*$ 视为\textbf{同时存在}的一个\textbf{人口}中具有不同特异性状态的家庭的价值与策略。
    \item 由于我们已经在整个网格上计算了 $V$，因此可以以非常低的成本加入异质性。
    \item 我们只需要思考如何\textbf{表示}和\textbf{模拟}一个家庭人口。
  \end{itemize}
\end{frame}

\begin{frame}{人口就是一个向量（或矩阵）}
  我们可以把\emph{人口}表示为定义在财富网格上的离散分布 $\lambda$：
  \[
    \lambda_i \;=\; \text{位于 } b_i \text{ 的家庭份额}.
  \]
  也就是说，正如 $V$ 一样，它是一个在离散化状态空间上的向量（或矩阵）。

  （注：矩阵也是向量。）
\end{frame}

\begin{frame}{总量矩}
  \begin{itemize}
    \item 总福利就是一个内积：$\bar{V} = \sum_i V(b_i) \, \lambda_i$。对 $\bar{C}$、$\bar{K}$ 等同样适用。
    \item 实际上，分布的\textbf{任意}矩（例如总资本、总消费等）都可以视为与 $\lambda$ 的内积。
  \end{itemize}
\end{frame}

\begin{frame}{三个阶段的前向与反向迭代}
  \footnotesize
  \setlength{\tabcolsep}{6pt}
  \renewcommand{\arraystretch}{1.45}
  \begin{tabular}{@{}l l l@{}}
  \toprule
  & \textbf{反向}（值函数） & \textbf{前向}（分布） \\
  \midrule
  \textbf{阶段 1}    & $V^{\mathrm{start}} = V^{\mathrm{pre\_inc}}\, \Pi^\top$
                      & $\lambda^{\mathrm{post}} = \lambda^{\mathrm{pre}}\, \Pi$ \\
  \textbf{阶段 2}    & $V^{\mathrm{pre\_inc}}(b^{\mathrm{end}}, z) = V^\text{pre\_cons}(R\, b^{\mathrm{end}} + z,\, z)$
                      & $(b^{\mathrm{end}}, z) \to (R\, b^{\mathrm{end}} + z,\, z)$ \\
  \textbf{阶段 3}    & $V^\text{pre\_cons}(b, z) = \max\limits_{b^{\mathrm{end}}} u(b - b^{\mathrm{end}}) + V^{\mathrm{end}}(b^{\mathrm{end}}, z)$
                      & $(b, z) \to \big(b - c^\star(b, z),\, z\big)$ \\
  \midrule
  \textbf{动态边界条件}   & $V^{\mathrm{end}_t} = \beta\, V^{\mathrm{start}}_{t+1}$
                      & $\Lambda^\text{start}_t = \Lambda^\text{end}_{t+1}$ \\
  \textbf{稳态}   & $V^{\mathrm{end}} = \beta\, V^{\mathrm{start}}$
                      & $\Lambda^\text{start} = \Lambda^\text{end}$ \\
  \bottomrule
  \end{tabular}

  \vspace{0.8em}

  \footnotesize
  反向：$V^{\mathrm{end}} \to V^\text{pre\_cons} \to V^{\mathrm{pre\_inc}} \to V^{\mathrm{start}}$。

  前向：$\Lambda^\text{start} \to \Lambda^\text{pre\_inc} \to \Lambda^\text{pre\_cons} \to \Lambda^\text{end}$。
\end{frame}

\begin{frame}[fragile]{阶段 1：马尔可夫收入冲击}
  前向：
  \begin{lstlisting}
  # λ is an N_b × N_z matrix: rows index wealth, columns index z state.
  # Markov along the z dimension is a single matrix multiplication.
  result = λ * P_z
  \end{lstlisting}

  \begin{itemize}
    \item 在 $(b_i, z_j)$ 处的质量以权重 $\Pi_{j j'}$ 流向 $(b_i, z_{j'})$。
    \item 沿一个维度的马尔可夫演化就是一次矩阵乘法。
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{\texttt{simulate\_population}}
  \begin{lstlisting}
  function simulate_population(b_inds_new, params, λ_0; T = 200)
      path, cur = [copy( λ_0)], copy( λ_0)
      for t in 1:T
          cur = income_shock_forward(cur, params.P_z)                  # Stage 1
          cur = income_forward(cur, params)                            # Stage 2
          cur = consumption_saving_forward(cur, b_inds_new, params)    # Stage 3
          push!(path, copy(cur))
      end
      return path
  end
  \end{lstlisting}
  前向模拟同样不过是各阶段前向模拟函数的\textbf{函数复合}。
\end{frame}


\begin{frame}{求解稳态}
  \begin{enumerate}
    \item 猜测 $v$ 与 $\lambda$。
    \item 反向迭代 $v$ 直到收敛到 $V$。
    \item 前向模拟 $\lambda$ 直到收敛到 $\Lambda$。
  \end{enumerate}
  （注：$\lambda$ 收敛所需的条件比 $V$ 的更微妙。）
\end{frame}
