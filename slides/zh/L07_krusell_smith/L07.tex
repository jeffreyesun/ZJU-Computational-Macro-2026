\section[维度灾难]{维度灾难（Curse of Dimensionality）}
% =============================================================

\begin{frame}{设定}

\textbf{总量状态}（aggregate state）。全要素生产率（total factor productivity, TFP）$Z_t \in \{Z_{\text{bad}}, Z_{\text{good}}\}$ 服从转移矩阵为 $P_Z$ 的马尔可夫链（Markov chain）。

\smallskip

\textbf{个体状态}（idiosyncratic state）。$(z_{it}, b_{it})$。生产率 $z_{it} \in \{z_{\text{lo}}, z_{\text{hi}}\}$ 服从转移矩阵为 $P_z$ 的马尔可夫链；财富为 $b_{it}$。

\smallskip

\textbf{技术。}带 TFP $Z_t$ 的柯布--道格拉斯（Cobb--Douglas）形式：
\[
  Y_t = Z_t\, K_t^{\alpha} L^{1-\alpha}.
\]

\smallskip

$L$ 取为常数，等于 $z$ 的平稳均值。

\end{frame}

\begin{frame}{价格}

由 $Z_t, K_t, L$ 给出的租金率与工资：
\[
  r(Z_t, K_t) = \alpha\, Z_t\, (K_t/L)^{\alpha-1} - \delta,
\]
\[
  w(Z_t, K_t) = (1-\alpha)\, Z_t\, (K_t/L)^{\alpha}.
\]

\smallskip

问题：家庭对价格的预测既\textbf{影响}未来的 $K_{t+1}$，又\textbf{依赖}于未来的 $K_{t+1}$。

\end{frame}

\begin{frame}{$K_{t+1}$ 依赖于 $\Lambda_t$}

\[
  K_{t+1} = \int b_i'(b_i, z_i;\, Z_t, K_t)\, \mathrm d\Lambda_t(b_i, z_i).
\]

\smallskip

策略 $b_i'(b_i, z_i;\, Z_t, K_t)$ 依赖于价格预测。

价格 $r_{t+s}, w_{t+s}$ 的预测又通过 $K_{t+s}$ 依赖于 $\Lambda_{t+s}$。

\smallskip

\textbf{注意：}来自不同 $\Lambda_t$ 的相同 $K_t$ 可以蕴含不同的 $K_{t+1}$。

\end{frame}

\begin{frame}{贝尔曼方程}

\textbf{分布} $\Lambda$ 成为一个\textbf{状态变量}（state variable）：
\[
  V(b_i, z_i; \Lambda, Z) = \max_{c_i, b_i'} u(c_i) + \beta\, \E\!\bigl[V(b_i', z_i'; \Lambda', Z') \mid z_i, Z\bigr],
\]

\end{frame}

\begin{frame}{$\Lambda$ 是无穷维的}

$\Lambda_t$ 是定义在所有 $(b, z)$ 上的\textbf{分布}。

\smallskip

``真实''的 $\Lambda_t$ 是无穷维的。

\smallskip

即便把 $\Lambda_t$ 离散化为一个矩阵，\textbf{每一个网格点都是一个状态变量}。

取 $N_w = 80$ 个财富网格点与 $N_z = 2$ 个生产率水平，网格共有 160 个点。

\smallskip

一个覆盖 $\Lambda$ 所有可能取值的 $V$ 网格：$5^{160-1}$ 个单元，约 $10^{111}$。

\end{frame}

\begin{frame}{对 $V(\ldots\ ;\ \ldots,\ \Lambda)$ 的可处理参数化}

\textbf{(a) 压缩 $\Lambda$。}（Krusell--Smith。）用 $\Lambda_t$ 的若干矩（moments）来代替整个分布。

\smallskip

\textbf{(b) 压缩表示。}（nVFI。）用一族灵活的函数（例如\textbf{神经网络}（neural network））来参数化 $V(\Lambda, \cdot)$。

\end{frame}

% =============================================================
\section[K--S 方法]{Krusell--Smith 方法}
% =============================================================

\begin{frame}{微观基础：有限理性}

Krusell 与 Smith（1998）：假设\textbf{家庭}只观察到 $\Lambda_t$ 的若干矩，而非整个分布。

\smallskip

即：求解 $V(b, z; Z, K)$ 的模型，而非 $V(b, z; Z, \Lambda)$ 的模型。

\end{frame}

\begin{frame}{近似运动律}

假设家庭对 $K$ 使用如下预测规则（forecasting rule）：
\[
  \log K_{t+1} = a_0 Z_t + a_1 Z_t\, \log K_t.
\]

\textbf{每个 $Z$ 状态两个系数。}对于 2 状态的 $Z$，共四个数。

\textbf{对数线性。}K--S 发现这一形式拟合得非常好。

\end{frame}

\begin{frame}{有限理性下的贝尔曼方程}

在状态中用 $K$ 替换 $\Lambda$：
\[
  V(b_i, z_i, Z, K) = \max_{c_i, b_i'} u(c_i) + \beta\, \E\bigl[V(b_i', z_i', Z', K')\bigr],
\]
约束为
\[
  c_i + b_i' = (1 + r(Z, K))\, b_i + w(Z, K)\, z_i,
\]
家庭的\emph{信念}（belief）为
\[
  K' = \exp(a_0(Z) + a_1(Z) \log K),
\]
$Z'$ 从 $P_Z$ 中抽取。

\end{frame}

\begin{frame}{由五个阶段构成的链}

每一期内，按时间顺序：
\begin{enumerate}
  \item \texttt{z\_shock}：$z_{it} \to z_i^{t+1}$（\texttt{:z} 上的马尔可夫转移）。
  \item \texttt{income}：$b_i' \gets (1+r) b_i + w\, z_i$，其中 $(r, w)$ 由 $(Z, K)$ 决定。
  \item \texttt{savings}：选择 $b_i^{t+1}$。
\end{enumerate}
矩阶段：\texttt{K\_evolve}：$K_t \to K_{t+1}$。
总量冲击：\texttt{Z\_shock}：$Z_t \to Z_{t+1}$。

\smallskip
\begin{center}\small
\texttt{z\_shock} $\circ$ \texttt{income} $\circ$ \texttt{savings}.
\end{center}
\end{frame}

\begin{frame}{K--S 外循环}

\begin{enumerate}
  \item \textbf{选取}初始的近似运动律（approximate law of motion, ALM）$(a_0^{(0)}, a_1^{(0)})$。
  \item 在当前 ALM 下\textbf{求解}四维家庭 VFI。
  \item 在实现的 $\{Z_t\}$ 路径上\textbf{模拟}一个长时段经济，并用\emph{真实的}聚合方式 $K_t = \int b\, \mathrm d\Lambda_t$ 计算 $K_t$。
  \item 按 $Z$ 分组，对 $\log K_{t+1}$ 关于 $\log K_t$ 做 OLS 回归，\textbf{重新拟合} $(a_0, a_1)$。
  \item \textbf{迭代}直到 ALM 系数不再变化。
\end{enumerate}

\end{frame}

\section[实现 K--S]{实现 K--S}
% =============================================================

\begin{frame}{参数}

\textbf{经典 K--S 校准}（1998）：

\begin{itemize}
  \item $\beta = 0.96$，$\gamma = 1$（对数效用），$\alpha = 0.36$，$\delta = 0.025$。
  \item $Z \in \{0.99, 1.01\}$，$P_Z$ 具有较强持续性（对角元为 $0.875$）。
  \item $z \in \{0.07, 1.0\}$。
\end{itemize}

\smallskip

\textbf{网格}（为教学速度考虑，取较小规模）：

\begin{itemize}
  \item $N_w = 80$，在 $[0, 80]$ 上对数等距。
  \item $N_K = 5$，在 $[10.5, 14.5]$ 上线性等距。
  \item 单元数：$80 \times 2 \times 2 \times 5 = 1600$。
\end{itemize}

\end{frame}

\begin{frame}{准备：在 $Z = 1$ 处确定 K 网格中心}

在 K--S 迭代开始之前，我们需要：

\begin{itemize}
  \item $\bar K$，K 网格的中心（即家庭的信念集中心）。
  \item $\Lambda_0$，模拟所用的初始财富分布。
\end{itemize}

\smallskip

两者都来自\textbf{在 $Z = 1$ 处的确定性 Aiyagari 稳态}---即第 4 讲的 tatonnement 在将 $Z$ 固定为 $\{Z_{\text{bad}}, Z_{\text{good}}\}$ 中点时的应用。此时状态中无 $Z$ 也无 $K$ 的二维家庭模块对 $K$ 单独闭合循环。

\smallskip

在此网格上，tatonnement 在约 10 次迭代内收敛到 $\bar K \approx 13$。

\end{frame}

\begin{frame}[fragile]{阶段 1--2：z\_shock 与 income \\ Stage 1--2}
\begin{lstlisting}
z_shock = MarkovStage(layout; axis=:z, transition=p.P_z)

income = WealthChangeStage(layout;
    wealth_post=(cell; env) -> begin
        Z  = p.Z_vals[cell.Z_idx]
        pr = ks_prices(cell.K, Z, p)
        (1 + pr.r) * cell.wealth + pr.w * cell.z
    end,
)
\end{lstlisting}

\textbf{注。}income 闭包从\emph{单元坐标}中读取 $(Z, K)$，而不是从 \texttt{env} 中读取。因此在单次求解\emph{内部}，价格也会随单元变化。

\end{frame}

\begin{frame}[fragile]{阶段 3：ConsumptionSavings \\ Stage 3}
\begin{lstlisting}
savings = ConsumptionSavingsStage(layout;
    β=p.β,
    utility=(cell, c; env) -> u_crra(c, Val(p.γ)),
    monotone_search=:divide_conquer,
)
\end{lstlisting}

\vspace{0.4em}

\begin{itemize}
  \item 与 Aiyagari / 空间模型中所用阶段相同。
  \item 效用按单元定义---但在这里只依赖于 $c$。
  \item 分治搜索利用了关于财富的单调性。
\end{itemize}

\end{frame}

\begin{frame}[fragile]{阶段 4--5：K\_evolve 与 Z\_shock \\ Stage 4--5}
\begin{lstlisting}
K_evolve = WealthChangeStage(layout;
    wealth_axis=:K,
    wealth_post=(cell; env) -> begin
        a0 = env.a0[cell.Z_idx]
        a1 = env.a1[cell.Z_idx]
        exp(a0 + a1 * log(cell.K))
    end,
)

Z_shock = MarkovStage(layout; axis=:Z_idx, transition=p.P_Z)
\end{lstlisting}

\textbf{K\_evolve} 是 \texttt{:K} 上的\emph{确定性}质量重分配阶段，从 \texttt{env} 中读取 ALM 系数。\texttt{WealthChangeStage} 并不只用于财富---它可在任何连续维度上重分配质量；此处我们将其用于 \texttt{:K}。

\end{frame}

\begin{frame}[fragile]{组合；声明矩}
\begin{lstlisting}
chain = z_shock ∘ income ∘ savings ∘ K_evolve ∘ Z_shock

hh = define_moments!(chain;
    K_supplied=at_end(integrand=:wealth, reduce=sum))
\end{lstlisting}

\vspace{0.4em}

\begin{itemize}
  \item \texttt{K\_supplied} 对期末 $\Lambda$ 上的 $b$ 求和---这正是 K--S 迭代中出现的积分。
  \item 用于在 $Z = 1$ 处的确定性 Aiyagari 稳态求解，\emph{以及} K--S 循环过程中的诊断。
\end{itemize}

\end{frame}

\begin{frame}[fragile]{内层求解：在候选 ALM 下的 VFI}
\begin{lstlisting}
a0_init = [log(K_bar), log(K_bar)]
a1_init = [0.0,        0.0]
env0    = make_env(hh; a0=a0_init, a1=a1_init)

                                   lambda_tol=1e-5,
                                   lambda_maxiter=50_000)
\end{lstlisting}

\textbf{初始猜测。}一个\emph{常数} ALM：对两个 $Z$ 状态都有 $K_{t+1} = \bar K$（即 $a_1 \equiv 0$，$a_0 \equiv \log \bar K$）。

\textbf{开销。}在 1600 单元网格上约 400 次内层 VFI 迭代。
\end{frame}

\begin{frame}{模拟：为什么要单独构造一条链}

模拟时我们想要的是\emph{实现的}$(Z_t, K_t)$，而不是家庭的信念。

\smallskip

原链中的 K\_evolve 与 Z\_shock 编码的是\emph{信念}：它们会沿 ALM 推进 $K$，沿 $P_Z$ 推进 $Z$，与实际实现无关。

\smallskip

\textbf{解决办法。}构造第二条链---只包含前三个阶段（\texttt{z\_shock} $\circ$ \texttt{income} $\circ$ \texttt{savings}）---并从外部驱动其 $(Z, K)$。

\end{frame}

\begin{frame}{一次模拟步}

将 $\Lambda_t$ 维护为定义在 $(b, z)$ 上的\emph{二维}分布。每一期：
\begin{enumerate}
  \item 计算 $K_t = \sum_{b, z} b\, \Lambda_t[b, z]$。
  \item 将 $\Lambda_t$ 提升为一个集中在 $(Z_t, K_t)$ 上的四维分布---份额在两个相邻的 K 网格点之间分摊。
  \item 让该四维分布通过模拟链（3 个阶段）。
  \item 对 $(Z, K)$ 做边缘化，得到 $(b, z)$ 上的 $\Lambda_{t+1}$。
  \item 从 $P_Z$ 中抽取 $Z_{t+1} \mid Z_t$。
\end{enumerate}

\end{frame}

\begin{frame}[fragile]{为模拟链做准备}
\begin{lstlisting}
hh_sim = ks_household_sim(p)
# Run backward on the sim chain with the converged V_ss as
# terminal value, so its savings policy is materialised.

\end{lstlisting}

\vspace{0.4em}

\begin{itemize}
  \item 模拟链不含 K\_evolve/Z\_shock，因此其 env 不需要 ALM 系数。
  \item \texttt{backward!} 利用内层求解得到的 $V^*$ 来填充储蓄策略。
  \item 之后 \texttt{ks\_simulate} 即可推动 $\Lambda$ 向前演化。
\end{itemize}

\end{frame}

\begin{frame}[fragile]{外循环}
\begin{lstlisting}
function ks_iterate(p, K_bar, Λ_0; maxiter=15)
    hh, hh_sim = ks_household(p), ks_household_sim(p)
    a0, a1 = [log(K_bar), log(K_bar)], [0.0, 0.0]
    for it in 1:maxiter
        _ss = solve_steady_state_given_env!(hh, make_env(hh; a0, a1); ...)
        backward!(hh_sim, copy(_ss.V), (;))
        sim = ks_simulate(hh_sim, p, T_sim; Λ_init=Λ_0, Z_init_idx=1)
        fit_b = regress_ALM(sim.K_path, sim.Z_idx_path, 1, window_fit)
        fit_g = regress_ALM(sim.K_path, sim.Z_idx_path, 2, window_fit)
        a0_n, a1_n = [fit_b.a0, fit_g.a0], [fit_b.a1, fit_g.a1]
        Δ = maximum(abs.(vcat(a0_n .- a0, a1_n .- a1)))
        Δ < 1e-3 && return (; a0, a1, sim, iters=it)
        a0 = 0.5 .* a0 .+ 0.5 .* a0_n
        a1 = 0.5 .* a1 .+ 0.5 .* a1_n
    end
end
\end{lstlisting}
\end{frame}

\begin{frame}{运行结果}

\textbf{收敛行为。}

\begin{itemize}
  \item 在 \textbf{11 次外循环}内收敛到 $\Delta < 10^{-3}$。
  \item 内层 VFI：每轮外循环约 400 次迭代，随接近不动点而减少。
  \item 预热后窗口内 $K$ 的均值：约 $13.70$。
\end{itemize}

\smallskip

将系数更新速率设为 $0.5$ 可以避免前几轮出现超调（overshooting）。

\end{frame}

% =============================================================
\section[精度]{精度之争（The Accuracy Debate）}
% =============================================================

\begin{frame}{收敛后的 ALM}

经过 11 次外循环后：
\[
  \log K_{t+1} = 0.18 + 0.93\, \log K_t \quad (Z = \text{bad}),
\]
\[
  \log K_{t+1} = 0.15 + 0.94\, \log K_t \quad (Z = \text{good}).
\]

\smallskip

$\log K_t$ 的系数非常接近（$0.93$ 对 $0.94$）；截距有所差异（$0.18$ 对 $0.15$），刻画了衰退与繁荣之间的水平移位。

\smallskip

仅四个数字就描述了 $K$ 在财富分布的所有状态下的动态。

\end{frame}

\begin{frame}{令人震惊的结果：$R^2 \approx 1$}

收敛后的 ALM 回归：

\begin{itemize}
  \item $Z = \text{bad}$：$R^2 = 0.99696$。
  \item $Z = \text{good}$：$R^2 = 0.99862$。
\end{itemize}

\smallskip

\textbf{如何理解。}$\log K_t$ 的线性函数\emph{基本上完全}解释了每个 $Z$ 状态内 $\log K_{t+1}$ 的变异。

\smallskip

\textbf{含义。}完整分布 $\Lambda_t$ 在均值 $K_t$ 之外不再提供可用的信息。只知道 $K_t$ 的家庭与知道整个 $\Lambda_t$ 的家庭预测得一样好。

\end{frame}

\begin{frame}{Den Haan 的批评}

$R^2$ 是对多期残差的平均---它可能掩盖\emph{系统性漂移}：一次很小的单步偏差在长期会累积放大。

\smallskip

\textbf{Den Haan（2010）。}给出了一种更尖锐的诊断。从相同的 $K_0$ 出发：

\begin{itemize}
  \item 每期通过聚合家庭决策计算真实的 $K_t$ 路径。
  \item 计算仅由 ALM 给出的路径：$K_{t+1} = \exp(a_0(Z_t) + a_1(Z_t) \log K_t)$，不做聚合。
\end{itemize}

在数百期上进行比较。

\end{frame}

\begin{frame}{我们看到的结果}

在本校准下，在预热后的 800 期里：
\[
  \max_t \left|\frac{K_t^{\text{ALM}} - K_t^{\text{true}}}{K_t^{\text{true}}}\right| \approx 0.81\%.
\]

\smallskip

仅由 ALM 给出的路径在整段模拟中与真实路径的偏差均在约 1\% 之内。

\smallskip

\textbf{诠释。}有限理性（bounded rationality）假设在此基本上是精确的。K--S 主体的预测与完全理性主体一样好---不仅平均如此，而且\emph{逐条轨迹}都成立。

\end{frame}

\begin{frame}{K--S 何时有效，何时失效}

\textbf{K--S 在以下情形有效：}

\begin{itemize}
  \item 储蓄关于财富近似线性。
  \item 主体较为耐心。
  \item 不存在显著的 MPC 异质性。
\end{itemize}

\textbf{K--S 可能失效的情形：}

\begin{itemize}
  \item 职业选择 / 广延边际进入。
  \item 厚尾收入冲击。
  \item 非 CRRA 的 $u(c)$（习惯形成、损失厌恶等）。
  \item 破产 / 违约。
\end{itemize}

\end{frame}

\begin{frame}{线性 ALM 看不见什么}

ALM 把 $\Lambda_t$ 压缩成一个标量。那些在不显著移动 $K_t$ 的前提下，在不同 MPC 的主体之间重新分配财富---或在不同机制之间（受约束 vs.\ 不受约束）转移质量---的结构性变化，对 ALM 而言是\emph{不可见的}。

\smallskip

在基准校准下，这些变化恰好对 $K_{t+1}$ 影响很小。在更丰富的模型中则未必如此。

\smallskip
