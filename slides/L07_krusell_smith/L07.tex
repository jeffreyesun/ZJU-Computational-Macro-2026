\section{Krusell-Smith}
% =============================================================

\begin{frame}{Setup: Aiyagari with Aggregate TFP Shocks}
  Aggregate TFP $Z_t \in \{Z_{\text{bad}}, Z_{\text{good}}\}$ Markov  with $\Pi_Z$.

  Idiosyncratic state: $(z_{it}, b_{it})$
  \begin{itemize}
    \item Productivity $z_{it} \in \{z_{\text{lo}}, z_{\text{hi}}\}$ Markov with $\Pi_z$
    \item Wealth $b_{it}$.
  \end{itemize}

  Technology: Cobb--Douglas with TFP $Z_t$:
  \[
    Y_t = Z_t\, K_t^{\alpha} L^{1-\alpha}.
  \]
  $L$ is assumed constant at the stationary mean of $z$.
\end{frame}

\begin{frame}{Prices}
  Gross interest rate and wage given by $Z_t, K_t, L$,
  \[
    R(Z_t, K_t) = 1 + \alpha\, Z_t\, (K_t/L)^{\alpha-1} - \delta,
  \]
  \[
    w(Z_t, K_t) = (1-\alpha)\, Z_t\, (K_t/L)^{\alpha}.
  \]

  Problem: Household price forecasts both \textbf{affect} and \textbf{depend on} future $K_{t+1}$.
\end{frame}

\begin{frame}{$K_{t+1}$ depends on $\Lambda_t$}
  \[
    K_{t+1} = \int b_i'(b_i, z_i;\, Z_t, K_t)\, \mathrm d\Lambda_t(b_i, z_i).
  \]

  Policy $b_i'(b_i, z_i;\, Z_t, K_t)$ depends on price forecasts.

  Price $R_{t+s}, w_{t+s}$ forecasts depend on $\Lambda_{t+s}$ through $K_{t+s}$.

  \textbf{Note:} Same $K_t$ from different $\Lambda_t$ can imply different $K_{t+1}$.
\end{frame}

\begin{frame}{Bellman Equation}
  \textbf{Distribution} $\Lambda$ becomes a \textbf{state variable}:
  \[
    V(b_i, z_i; \Lambda, Z) = \max_{c_i, b_i'} u(c_i) + \beta\, \E\!\bigl[V(b_i', z_i'; \Lambda', Z') \mid z_i, Z\bigr],
  \]
\end{frame}

\begin{frame}{$\Lambda$ is infinite-dimensional}
  $\Lambda_t$ is a \textbf{distribution} over every $(b, z)$.

  The ``true'' $\Lambda_t$ is infinite-dimensional.

  Even if we discretize $\Lambda_t$ into a matrix, \textbf{every gridpoint is a state variable}.

  With $N_b = 80$ wealth gridpoints and $\#z = 2$ productivity levels, grid has 160 points.

  A $V$ grid over possible $\Lambda$ values: $5^{160-1}$ cells. About $10^{111}$.
\end{frame}

\begin{frame}{Tractably Parametrizing $V(\ldots\ ;\ \ldots,\ \Lambda)$}
  Two ideas:

  \textbf{(a) Reduce $\Lambda$.} (Krusell-Smith.) Use moments of $\Lambda_t$ instead of the whole distribution.

  \textbf{(b) Reduce the representation.} (nVFI.) Parameterise $V(\Lambda, \cdot)$ by a flexible family of functions (e.g. a \textbf{neural network}).
\end{frame}

% =============================================================
\section[K--S Method]{Krusell--Smith Method}
% =============================================================

\begin{frame}{Microfoundation: Bounded Rationality}
  Krusell and Smith (1998): Assume the \textbf{household} only observes moments of $\Lambda_t$, not the whole distribution.

  i.e. solve a model of $V(b, z; Z, K)$, not $V(b, z; Z, \Lambda)$.
\end{frame}

\begin{frame}{The Approximate Law of Motion (ALM)}
  Assume household uses a forecast rule for $K$:
  \[
    \log K_{t+1} = a_0(Z_t) + a_1(Z_t) \log K_t.
  \]

  \textbf{Two coefficients per $Z$ state.} For 2-state $Z$, four numbers total.

  \textbf{Log-linear.} K--S find this is actually a very good fit.
\end{frame}

\begin{frame}{The Bounded-Rationality Bellman}
  Replace $\Lambda$ with $K$ in the state:
  \[
    V(b_i, z_i, Z, K) = \max_{c_i, b_i'} u(c_i) + \beta\, \E\bigl[V(b_i', z_i', Z', K')\bigr],
  \]
  subject to
  \[
    c_i + b_i' = R(Z, K)\, b_i + w(Z, K)\, z_i,
  \]
  with the agent's \emph{belief}
  \[
    K' = \exp(a_0(Z) + a_1(Z) \log K),
  \]
  and $Z'$ drawn from $\Pi_Z$.
\end{frame}

\begin{frame}{The Behavioural Chain (Five Stages)}
  Assuming the approximate law of motion (ALM), we can solve the model by simply adding two Stages to the household block:
  \begin{enumerate}
    \item \texttt{z\_shock}: $z_{it} \to z_i^{t+1}$ (Markov on \texttt{:z}).
    \item \texttt{income}: $b_i' \gets R\, b_i + w\, z_i$, with $(R, w)$ from $(Z, K)$.
    \item \texttt{savings}: pick $b_i^{t+1}$.
    \item \texttt{K\_evolve}: $K_t \to K_{t+1}$.
    \item \texttt{Z\_shock}: $Z_t \to Z_{t+1}$.
  \end{enumerate}

  \begin{center}\small
  \texttt{z\_shock} $\circ$ \texttt{income} $\circ$ \texttt{savings} $\circ$ \texttt{K\_evolve} $\circ$ \texttt{Z\_shock}.
  \end{center}

  Solving this chain gives us household \textbf{behaviour} given $a_0(\cdot), a_1(\cdot)$, and allows us to simulate the \textbf{true} model forward.
\end{frame}

\begin{frame}{K--S Outer Loop}
  \begin{enumerate}
    \item \textbf{Pick} an initial ALM $(a_0^{(0)}, a_1^{(0)})$.
    \item \textbf{Solve} the 4D household VFI given the current ALM.
    \item \textbf{Simulate} a long economy under realised $\{Z_t\}$ and \emph{true} aggregation $K_t = \int b\, \mathrm d\Lambda_t$.
    \item \textbf{Refit} $(a_0, a_1)$ by OLS of $\log K_{t+1}$ on $\log K_t$, separately per $Z$.
    \item \textbf{Iterate} until ALM coefficients stop moving.
  \end{enumerate}
\end{frame}

\section{Implementing Krusell--Smith}
% =============================================================

\begin{frame}{Parameters}
  \textbf{Krusell--Smith calibration} (1998):
  \begin{itemize}
    \item $\beta = 0.96$, $\gamma = 1$ (log utility), $\alpha = 0.36$, $\delta = 0.025$.
    \item $Z \in \{0.99, 1.01\}$ with $\Pi_Z$ persistent ($0.875$ on-diagonal).
    \item $z \in \{0.07, 1.0\}$.
  \end{itemize}

  \textbf{Grids} (small for example):
  \begin{itemize}
    \item $N_b = 80$ on log-spaced $[0, 80]$.
    \item $N_K = 5$ on linear $[10.5, 14.5]$.
    \item Cell count: $80 \times 2 \times 2 \times 5 = 1600$.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{State Layout}
  \begin{lstlisting}
  layout = StateLayout(
      StateAxis(:wealth, continuous_grid(p.b_min, p.b_max; length=p.N_b, spacing=:log)),
      StateAxis(:z,      p.z_grid),
      StateAxis(:Z_idx,  [1, 2]),
      StateAxis(:K,      continuous_grid(p.K_min, p.K_max; length=p.N_K)),
  )
  \end{lstlisting}

  \begin{itemize}
    \item Two continuous axes (\texttt{:wealth}, \texttt{:K}), two categorical (\texttt{:z}, \texttt{:Z\_idx}).
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Stages 1--2: z\_shock and income \\ Stage 1--2}
  \begin{lstlisting}
  z_shock = MarkovStage(layout; axis=:z, transition=p.Π_z)

  income = WealthChangeStage(layout; wealth_post=(cell; env) -> begin
      Z = p.Z_vals[cell.Z_idx]
      (;R, w) = ks_prices(cell.K, Z, p)
      R * cell.wealth + w * cell.z
  end)
  \end{lstlisting}

  \textbf{Note.} This reads $(Z, K)$ \emph{from the cell coordinates}.
\end{frame}

\begin{frame}[fragile]{Stage 3: ConsumptionSavings \\ Stage 3}
  \begin{lstlisting}
  savings = ConsumptionSavingsStage(layout; β=p.β,
      utility=(cell, c; env) -> u_crra(c, Val(p.γ)),
      monotone_search=:divide_conquer)
  \end{lstlisting}

  \begin{itemize}
    \item Divide-and-conquer search exploits monotonicity in wealth.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Stages 4--5: K\_evolve and Z\_shock \\ Stage 4--5}
  \begin{lstlisting}
  K_evolve = WealthChangeStage(layout; wealth_axis=:K,
      wealth_post=(cell; env) ->
          exp(env.a0[cell.Z_idx] + env.a1[cell.Z_idx] * log(cell.K)))

  Z_shock = MarkovStage(layout; axis=:Z_idx, transition=p.Π_Z)
  \end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Build Household Block}
  \begin{lstlisting}
  chain = z_shock ∘ income ∘ savings ∘ K_evolve ∘ Z_shock

  hh = define_moments!(chain;
      K_supplied=at_end(integrand=:wealth, reduce=sum))
  \end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Inner solve: VFI Given $a_0(\cdot), a_1(\cdot)$}
  \begin{lstlisting}
  a0_init = [log(K_bar), log(K_bar)]
  a1_init = [0.0, 0.0]
  env0 = make_env(hh_4d; a0=a0_init, a1=a1_init)

  res4d = solve_steady_state_given_env!(hh_4d, env0; lambda_tol=1e-5, lambda_maxiter=50_000)
  \end{lstlisting}

  \textbf{First guess.} A \emph{constant} ALM: $K_{t+1} = \bar K$ for both $Z$ states ($a_1 \equiv 0$, $a_0 \equiv \log \bar K$).
\end{frame}

\begin{frame}{Simulation Household Block}

  When we simulate we want \emph{realised} $(Z_t, K_t)$, not the agent's belief.

  The behavioural chain's K\_evolve and Z\_shock encode the \emph{belief}, not the true law of motion for $K$.

  The true chain has just the first three stages (\texttt{z\_shock} $\circ$ \texttt{income} $\circ$ \texttt{savings}) and $Z, K$ are updated in an \textbf{outer loop}.
\end{frame}

\begin{frame}{Simulating the True Model}
  Each period:
  \begin{enumerate}
    \item Compute $K_t = \sum_{b, z} b\, \Lambda_t[b, z]$.
    \item Lift $\Lambda_t$ to a 4D distribution, adding $(Z_t, K_t)$ as ``idosyncratic'' state variables.
    \item Simulate the 4D distribution using the behavioural chain (5 stages).
    \item Marginalise out $(Z, K)$ to get $\Lambda_{t+1}$ on $(b, z)$.
    \item Draw $Z_{t+1} \mid Z_t$ from $\Pi_Z$.
  \end{enumerate}
\end{frame}

\begin{frame}[fragile]{Refit the ALM by OLS}
  \begin{lstlisting}
  function regress_ALM(K_path, Z_idx_path, z_idx, window)
      rows = [t for t in window if Z_idx_path[t] == z_idx]
      x = log.(K_path[rows])
      y = log.(K_path[rows .+ 1])
      # OLS: y = a0 + a1 * x ; R² = 1 - SS_res/SS_tot
      ...
      return (; a0, a1, R², n=length(rows))
  end

  fit_bad  = regress_ALM(sim0.K_path, sim0.Z_idx_path, 1, window_fit)
  fit_good = regress_ALM(sim0.K_path, sim0.Z_idx_path, 2, window_fit)
  \end{lstlisting}
\end{frame}

\begin{frame}[fragile]{The Outer Loop}
  (Too long for the slide; see notebook.)
  
  \begin{lstlisting}
  function ks_iterate(p::KSParams, K_bar, Λ_0; update_speed=0.5, rtol=1e-3, maxiter=20, T_sim=2000, T_burn=500)
      hh_4d = behavioural_alm_household(p)
      a0 = [log(K_bar), log(K_bar)]
      a1 = [0.0, 0.0]
      err, iters = Inf, 0
      while err >= rtol
          env = make_env(hh_4d; a0, a1)
          solve_steady_state_given_env!(hh_4d, env; lambda_tol=1e-5, lambda_maxiter=50_000)

          sim = ks_simulate(hh_4d, p; T=T_sim, Λ_init=Λ_0, Z_init=1)
          window_fit = (T_burn + 1):(T_sim - 1)
          fit_bad  = regress_ALM(sim.K_path, sim.Z_idx_path, 1, window_fit)
          fit_good = regress_ALM(sim.K_path, sim.Z_idx_path, 2, window_fit)

          a0_new = [fit_bad.a0, fit_good.a0]
          a1_new = [fit_bad.a1, fit_good.a1]
          err = maximum(abs.(vcat(a0_new .- a0, a1_new .- a1)))

          a0 = (1 - update_speed) .* a0 .+ update_speed .* a0_new
          a1 = (1 - update_speed) .* a1 .+ update_speed .* a1_new
          iters += 1
      end
      return (; a0, a1, sim, fit_bad, fit_good, iters)
  end
  \end{lstlisting}
\end{frame}
