\section[nVFI]{Neural Value Function Iteration (nVFI)}
% =============================================================

\begin{frame}{Neural Value Function Iteration (nVFI)}
  Broadly:
  \begin{enumerate}
    \item Parameterize $\Vend(\xend; \Lstart, Z)$ using a neural network with params $\theta$
    \item Sample from $\Lstart$ (somehow)
    \item For each $\Lstart$ compute the \textbf{outer Bellman residual} (defined below)
    \item Update $\theta$ and repeat from Step 2 until convergence
  \end{enumerate}
\end{frame}

\begin{frame}{Outer Bellman Residual}
  Consider the Krusell-Smith model.

  Let $\Vendhat_\theta(\xend; \Lstart, Z)$ be a $\theta$-parameterized neural network.

  Informally, we want to satisfy as closely as possible,
  \[
    \Vendhat_\theta(\xend; \Lstart, Z) - \E_{Z'}\psq{\Vstarthat\p{\xend; \Lstart{}', Z'}\mid Z} = 0,
  \]
  where $\Vstarthat(\xend;\Lstart{}',Z')$ is computed by \textbf{lookahead}, given $\Vendhat_\theta, Z'$.
\end{frame}

\section[Conventions]{Conventions and Timing}

\begin{frame}{Conventions}
  \begin{enumerate}
    \item Household states do not change between the end of one period and the start of the next:
    \[
      \xend_{it} = \xstart_{i,t+1}\quad\text{and}\quad\Lend_{it} = \Lstart_{i,t+1}.
    \]
    \item Aggregate shocks occur \textbf{between} household periods,
    \begin{itemize}
      \item i.e. between $\Vend(\xend; \Lstart, Z)$ and $\Vstart(\xstart; \Lstart{}', Z')$
    \end{itemize}
    \item A value function \textbf{slice} or \textbf{guess} depending only on \textbf{idiosyncratic} state is denoted by tilde,
    \[
      \Vstarttil(\xstart), \quad\text{but}\quad \Vstart(\xstart; \Lstart, Z).
    \]
    \begin{itemize}
      \item i.e. symbols such as $\Vstarttil$ represent \textbf{function-valued variables}. More details below.
    \end{itemize}
  \end{enumerate}
\end{frame}

\begin{frame}{Conventions: Dependence on $\Lstart$}
  \begin{itemize}
    \item The model is deterministic within the household period.
    \item Thus, $\Lend$ is uniquely determined by $\Lstart$.
    \item Thus, it is \textbf{theoretically valid} to write $\Vend(\xend; \Lstart, Z)$.
  \end{itemize}
  \textbf{Note:} I am \textbf{not} saying anything about how to compute or represent $\Vend$ on the computer yet!

  This is purely the statement that $\Vend(\xend; \Lstart, Z)$ is \textbf{well-defined}.
\end{frame}

\begin{frame}{Conventions: Partial Application of $\Vend$}
  \begin{itemize}
    \item Fixing $\Lstart$ and $Z$, we can construct,
    \[
      \Vendtil(\xend) \equiv \Vend(\xend; \Lstart, Z).
    \]
    \item In computer-science terms, this is a \textbf{partial application} of $\Vend$ at $(\Lstart, Z)$.
    \item Let us write this as an operator taking a \textbf{function} to a \textbf{function}
    \begin{align*}
      &\CV(\Lstart, Z) = \Vendtil
    \end{align*}
    \item Seems abstract, but for us $\CV$ is just a function from a \textbf{matrix and a scalar} to a \textbf{matrix}.
    \item Preview: $\CV$ is the object we will attempt to approximate using a \textbf{neural network}.
  \end{itemize}
\end{frame}

\begin{frame}{Continuation Value Operator}
  \begin{itemize}
    \item $CV$ is the \textbf{continuation value operator},
    \begin{align*}
      &\CV(\Lstart, Z) = \Vendtil \equiv \Vend(\cdot; \Lstart, Z)\\
    \end{align*}
    \item It returns a \textbf{function}. That is, we may write,
    \[
      \CV(\Lstart, Z)(\xend) \equiv \Vend(\xend; \Lstart, Z).
    \]
  \end{itemize}
  My view is that the \textbf{continuation value operator} $\CV$ is the natural extension of the value function to HA models with aggregate shocks.
\end{frame}

\section[CVIAYN]{Continuation Value Is All You Need}
% =============================================================

\begin{frame}{Continuation Value Is All You Need}
  Suppose we have a model: within-period solver $\Phi$, continuation value operator $\CV$, aggregate shock process $\Pi$, and initial aggregate state $(\Lstart, Z)$.
  
  Then we can simulate the model forward:
  \begin{columns}
    \begin{column}[T]{0.5\textwidth}
      \begin{enumerate}
        \item $\Vendtil = \CV(\Lstart, Z)$
        \item $(\Vstarttil, \Lend) = \Phi(\Vendtil, \Lstart, Z)$
        \item Draw $Z' \sim \Pi(\cdot \mid Z)$
        \item $(\Lstart{}', Z') \equiv (\Lend, Z')$
      \end{enumerate}
    \end{column}
    \begin{column}[T]{0.5\textwidth}
      \begin{itemize}
        \item[] (Project continuation value)
        \item[] (Simulate to end of period)
        \item[] (Draw aggregate shock)
        \item[] (Construct start of next period state)
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Lookahead Given $\CV$}
  Given $\CV$, we can also perform \textbf{lookahead}.
  \begin{columns}
    \begin{column}[T]{0.5\textwidth}
      \begin{enumerate}
        \item $\Vendtil = \CV(\Lstart, Z)$
        \item $(\Vstarttil, \Lend) = \Phi(\Vendtil, \Lstart, Z)$
        \item Draw $Z' \sim \Pi(\cdot \mid Z)$
        \item $(\Lstart{}', Z') \equiv (\Lend, Z')$
        \item $\Vendtil{}' = \CV(\Lstart{}', Z')$
        \item $(\Vstarttil{}', \Lend{}') = \Phi(\Vendtil{}', \Lstart{}', Z')$
        \item $\Vendtil{}^\text{,implied} = \E_{Z'}[\Vstarttil{}' \mid Z]$
      \end{enumerate}
    \end{column}
    \begin{column}[T]{0.5\textwidth}
      \begin{itemize}
        \item[] (Project continuation value)
        \item[] (Simulate to end of period)
        \item[] (Draw aggregate shock)
        \item[] (Construct start-of-next-period state)
        \item[] (Project end-of-next-period value)
        \item[] (Iterate back to start-of-next-period value)
        \item[] ($\Vendtil$ should be equal to this)
      \end{itemize}
    \end{column}
  \end{columns}

  We will be using this to generalize VFI.
\end{frame}

\begin{frame}{OLD Typical Bellman Equation and Operator}
  The typical Bellman equation defines $V$ via the relation,
  \[
    V(b, z) = \max_{c \in [0, b]} \; u(c) + \beta \, \E\!\left[\, V(b', z') \,\big|\, z \,\right]
  \]
  This can be thought of as a fixed point of the Bellman operator $\mathcal{T}$ which applies \textbf{lookahead},
  \[
    \mathcal{T}v(b, z) \equiv \max_{c \in [0, b]} \; u(c) + \beta \, \E\!\left[\, v(b', z') \,\big|\, z \,\right]
  \]
  So that the Bellman equation is,
  \[
    V = \mathcal{T}V.
  \]
  I define a similar \textbf{outer Bellman equation}.
\end{frame}

\begin{frame}{NEW Outer Bellman Equation}
  The true $\CV$ also satisfies a recursive equation, the \textbf{outer Bellman equation:}
  \[
    \Vend(\xend; \Lstart, Z) \;=\; \E_{Z'}\psq{\Vstarttil{}'(\xend)\,\Big|\, Z},
  \]
  where
  \begin{align*}
    \Vendtil &= \CV(\Lstart, Z) \\
    \Lend &= \Phi_\Lambda(\Lstart, \Vendtil, Z) \\
    Z' &\sim \Pi_Z(\cdot \mid Z) \\
    \Vendtil{}' &= \CV(\Lend, Z') \\
    \Vstarttil{}'  &= \Phi_V\bigl(\Lend, \Vendtil{}', Z'\bigr).
  \end{align*}
\end{frame}

\begin{frame}{The Outer Bellman Operator $\CL$}
  For any parametrized guess $\CV_\theta$, we can define the \textbf{outer Bellman operator} $\CL$ as:
  \[
    \CL\CV_\theta(\Lstart, Z)(\xend) \equiv \E_{Z'}\psq{\Vstarttil{}'(\xend)\,\Big|\, Z},
  \]
  where,
  \begin{align*}
    \Vendtil &= \CV_\theta(\Lstart, Z) \\
    \Lend &= \Phi_\Lambda(\Lstart, \Vendtil, Z) \\
    Z' &\sim \Pi_Z(\cdot \mid Z) \\
    \Vendtil{}' &= \CV_\theta(\Lend, Z') \\
    \Vstarttil{}'  &= \Phi_V\bigl(\Lend, \Vendtil{}', Z'\bigr).
  \end{align*}
\end{frame}

\begin{frame}{The Outer Bellman Residual}
  The true \textbf{continuation value operator} $\CV$ is defined by,
  \[
    \CL\CV = \CV.
  \]
  For any $\CV_\theta$, we can define the \textbf{outer Bellman residual} as,
  \[
    \CL\CV_\theta - \CV_\theta,
  \]
  and we can frame the problem of approximating $\CV$ via $\CV_\theta$ as,
  \[
    \min_\theta \norm{\CL\CV_\theta - \CV_\theta}^2.
  \]
\end{frame}

\section[Implementing CVIAYN]{Implementing CVIAYN}
% =============================================================

\begin{frame}{Training Loop}
  \begin{enumerate}
    \item Sample $(\Lstart, Z)$ from ergodic distribution by simulating forward given $\CV_\theta$.
    \item Compute $\CL V_\theta$ at each sample $(\Lstart_i, Z_i)$:
    \begin{enumerate}
      \item Network predicts $\Vendhat_\theta$ at $(\Lstart_i, Z_i)$.
      \item Simulate $\Lstart_i$ forward to $\Lend_i$ using $\Phi$.
      \item For each $Z'$, predict $\Vendhat{}'_\theta$ at $(\Lend_i, Z')$ using $\CV_\theta$.
      \item Iterate backward to get $\Vstart{}'$ using $\Phi$.
      \item Take expectation $\Vendhat_i \leftarrow \E_{Z'}\psq{\Vstart{}'\mid Z}$.
    \end{enumerate}
    \item Compute gradients of loss,
    \[
      \frac{\partial}{\partial\theta}\sum_i\norm{\CV_\theta(\Lstart_i, Z_i) - \Vendhat_i}^2,
    \]
    taking $\Vendhat_i$ as fixed labels, and update $\theta$.
  \end{enumerate}
\end{frame}

\begin{frame}{$\Phi$: Already Done}
  As covered in previous notebooks, the function,
  \[
    \Phi\p{\Vendtil, \Lstart, Z} = \p{\Vstarttil, \Lend},
  \]
  can be implemented with a chain of three HouseholdStages,
  \[
    \text{MarkovStage}\circ\text{WealthChange}\circ\text{ConsumptionSavings}
  \]
  acting on matrices representing the discretized idiosyncratic $(b, z)$ grid:
  \begin{itemize}
    \item Iterate backward to get $\Vstarttil$
    \item Iterate forward to get $\Lend$.
  \end{itemize}
\end{frame}

  \vspace*{-2.5em}
  Four sub-nets:
  \begin{enumerate}
    \item \texttt{pop\_agg\_net} $: (b, z) \to \R^{k}$ — per-cell embedding aggregated against $\Lstart$ to give the global moments $gms = \sum_{(b,z)} \Lambda(b, z) \cdot \texttt{pop\_agg\_net}(b, z) \in \R^k$.
    \item \texttt{V\_pre} $: (b, z) \to \R^m$ — cell-level features for the value head.
    \item \texttt{gm\_post} $: \R^k \to \R^m$ — projects $gms$ into the value head's mid-space; added cell-wise to \texttt{V\_pre}.
    \item \texttt{V\_post} $: \R^{m + N_Z} \to \R$ — final head; takes the summed features plus a one-hot of $Z$, outputs $\Vendhat$.
  \end{enumerate}

\begin{frame}[fragile]{Neural Network in Code}
  \begin{lstlisting}
  struct VNet{A, P, G, V}
      pop_agg_net :: A    # (b, z) → gm_dim         per-cell embedding for GM
      V_pre       :: P    # (b, z) → mid            per-cell features for V head
      gm_post     :: G    # gm_dim → mid            project GM to mid space
      V_post      :: V    # mid + N_Z → 1           combine + Z, read off V_end
  end

  function (net::VNet)(Λ::AbstractMatrix, Z_idx::Int, p::KSParams)
      pop_emb = net.pop_agg_net(HH_FEAT)          # (gm_dim, N_CELLS)
      gm      = pop_emb * Float32.(vec(Λ))        # (gm_dim,)   pop-weighted average
      h_pre   = net.V_pre(HH_FEAT)                # (mid, N_CELLS)
      h_gm    = net.gm_post(gm)                   # (mid,)
      h       = h_pre .+ h_gm                     # broadcast → (mid, N_CELLS)
      pred    = net.V_post(vcat(h, Z_ROWS[Z_idx])) # (1, N_CELLS)
      return reshape(pred, p.N_b, length(p.z_grid))
  end
  \end{lstlisting}
\end{frame}

\begin{frame}[fragile]{The Lookahead Operator $\CL V_\theta$}
  \begin{lstlisting}
  function lookahead(hh, vnet::VNet, Λ_start::AbstractMatrix, Z_idx::Int, p::KSParams)
      V_end_now = Float64.(vnet(Λ_start, Z_idx, p))
      env_now   = make_env(hh; ks_prices(integrate_K(hh, Λ_start), p.Z_vals[Z_idx], p)...)
      backward!(hh, V_end_now, env_now)                  # seats savings policy
      Λ_end = forward!(hh, Λ_start)

      V_end_label = zeros(Float64, p.N_b, length(p.z_grid))
      for Z_next in 1:N_Z
          V_start_next = inner_solve_backward(hh, vnet, Λ_end, Z_next, p)
          V_end_label .+= p.Π_Z[Z_idx, Z_next] .* V_start_next
      end
      return (; V_end_label, Λ_end)
  end
  \end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Training Loss}
  Treating \texttt{target} $\equiv \Vendhat_i$ as fixed labels, autodiff through \texttt{vnet}.

  \begin{lstlisting}
  function build_targets(hh, vnet::VNet, batch, p::KSParams)
      targets = NamedTuple{(:Λ, :Z_idx, :target), Tuple{Matrix{Float64}, Int, Matrix{Float32}}}[]
      for (Λ_start, Z_idx) in batch
          out = lookahead(hh, vnet, Λ_start, Z_idx, p)
          push!(targets, (; Λ=copy(Λ_start), Z_idx, target=Float32.(out.V_end_label)))
      end
      return targets
  end

  function bellman_residual_loss(vnet::VNet, targets, p::KSParams)
      L = 0f0
      for (Λ, Z_idx, target) in targets
          pred  = vnet(Λ, Z_idx, p)
          scale = max(var(target), 1f-6)             # normalise by target scale
          L    += sum((pred .- target).^2) / (length(pred) * scale)
      end
      return L / length(targets)
  end
  \end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Sampling $\Lstart$}
  MCMC-style: Simulate given the \emph{current} $V_\theta$, burn in, and subsample.

  \begin{lstlisting}
  function sim_one_step(hh, vnet::VNet, Λ::AbstractMatrix, Z_idx::Int, p::KSParams)
      V_end = Float64.(vnet(Λ, Z_idx, p))
      env   = make_env(hh; ks_prices(integrate_K(hh, Λ), p.Z_vals[Z_idx], p)...)
      backward!(hh, V_end, env)
      return forward!(hh, Λ)
  end

  function sample_ergodic(hh, vnet::VNet, p::KSParams; T=200, burn=100, every=2, rng=MersenneTwister(8421))
      Λ     = copy(Λ_ss)
      Z_idx = 1
      batch = Tuple{Matrix{Float64}, Int}[]
      for t in 1:T
          Λ     = sim_one_step(hh, vnet, Λ, Z_idx, p)
          Z_idx = sample_Z(Z_idx, p.Π_Z, rng)
          if t > burn && (t - burn) % every == 0
              push!(batch, (copy(Λ), Z_idx))
          end
      end
      return batch
  end
  \end{lstlisting}
\end{frame}

\begin{frame}[fragile]{The training loop}
  \begin{lstlisting}
  function train!(vnet::VNet, hh, p::KSParams; epochs=50, lr=1e-3, T=200, burn=100, every=4)
      opt  = Flux.setup(Adam(lr), vnet)
      hist = Float64[]
      rng  = MersenneTwister(1729)
      for epoch in 1:epochs
          batch   = sample_ergodic(hh, vnet, p; T, burn, every, rng=MersenneTwister(rand(rng, UInt32)))
          targets = build_targets(hh, vnet, batch, p)
          loss, grads = withgradient(vnet) do net
              bellman_residual_loss(net, targets, p)
          end
          Flux.update!(opt, vnet, grads[1])
          push!(hist, Float64(loss))
      end
      return hist
  end
  \end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Initialization and Pretraining}
  Pretrain on $V_{\mathrm{ss}}$ at perturbed $\Lambda_{\mathrm{ss}}$ and all $Z$.
  \begin{lstlisting}
  function perturb_Λ(Λ::AbstractMatrix, σ, rng)
      Λp = Λ .* (1 .+ σ .* randn(rng, size(Λ)))
      Λp = max.(Λp, 0)
      return Λp ./ sum(Λp)
  end

  function pretrain_to_ss!(vnet::VNet, V_ss_target, p::KSParams; epochs=300, lr=5e-3, n_Λ=6, σ=0.10, seed=4747)
      opt    = Flux.setup(Adam(lr), vnet)
      target = Float32.(V_ss_target)
      rng    = MersenneTwister(seed)
      hist   = Float64[]
      for epoch in 1:epochs
          Λ_cloud = [perturb_Λ(Λ_ss, σ, rng) for _ in 1:n_Λ]
          loss, grads = withgradient(vnet) do net
              L = 0f0
              for Λ in Λ_cloud, Z_idx in 1:N_Z
                  pred = net(Λ, Z_idx, p)
                  L += sum((pred .- target).^2)
              end
              L / (n_Λ * N_Z * length(target))
          end
          Flux.update!(opt, vnet, grads[1])
          push!(hist, Float64(loss))
      end
      return hist
  end
  \end{lstlisting}
\end{frame}

\begin{frame}{Generalizations: Models}
  \begin{itemize}
    \item General approach well-defined for any model for which you can implement $\Phi$.
    \item If you can't implement $\Phi$, then the issue does not lie with aggregate uncertainty.
    \item I provide \texttt{HouseholdStages.jl} to build high-performance household blocks.
    \item Can handle: spatial, multi-sector, OLG, search-matching, New Keynesian, frictional and durable assets (e.g. housing), rational inattention, Epstein-Zin preferences
  \end{itemize}  
\end{frame}

\begin{frame}{Generalizations: Segmented Markets}
  Segmented-markets models, such as those with multiple locations or sectors, present additional challenges.

  I develop:
  \begin{itemize}
    \item \textbf{Segmented Generalized Moments} to extend Han--Yang--E (2025) to model-consistently compress both segment (e.g. location) and aggregate data.
    \item \textbf{Lightweight Network Heads} method to speed up training of segmented-markets models with $n$ segments and $m$ household states from $O(nm)$ to effectively $O(n+m)$
  \end{itemize}
\end{frame}

\begin{frame}{Generalizations: Methods}
  The nVFI algorithm does not, in principle, depend on these implementation details:
  \begin{itemize}
    \item $\CV_\theta$ parameterization: I use neural net, but method just requires some trainable function approximator.
    \item Value function representation: I use a matrix over discretized idiosyncratic state, but you could use splines, Chebyshev polynomials, or even another neural network.
    \item Population representation: I use a matrix over discretized idiosyncratic state, but you could use moments, finite agents, etc.
    \item Within-period solver: I use largely grid-based methods, but you could use EGM, Reiter, etc.
  \end{itemize}
  As long as you can implement the \textbf{within-period solver} $\Phi$, this algorithm is well-defined.
\end{frame}

\begin{frame}{Relationship to HouseholdStages.jl}
  The deep connection between this and my work on Household Stages is the following:
  \begin{itemize}
    \item Discrete-time HA models have a \textbf{compositional structure} based on \textbf{functional composition}
    \item By contrast, continuous-time HA models have a compositional structure based on \textbf{additive composition}.
  \end{itemize}
  The paper and package on Household Stages explores the implications of this for \textbf{within-period solvers} $\Phi$

  Informed by this, nVFI generalizes the Bellman Equation in terms of $\CV$ and uses it for global solutions.
\end{frame}
