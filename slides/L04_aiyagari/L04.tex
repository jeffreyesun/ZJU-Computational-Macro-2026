\section{Stages and Outer Loops}
% =============================================================

\begin{frame}{Stages}
  \begin{itemize}
    \item Heterogeneous-agent macro models are made by combining \textbf{Stages} with \textbf{Outer Loops}.
    \item \textbf{Stages} are the basic building blocks. They define household problems, which are the \textbf{hard part}.
    \item Because \textbf{Stages} are conceptually \textbf{well-defined}, \textbf{modular}, and \textbf{composable}, complex macro models can be built by simply combining \textbf{Stages}.
  \end{itemize}
\end{frame}

\begin{frame}{Stages Solve the Household Block}
  \begin{itemize}
    \item A \textbf{Stage} defines how households make decisions given their \textbf{environment}: \textbf{prices} and \textbf{parameters}.
    \item \textbf{Stages} compose into the \textbf{household block}: value functions and populations given prices and parameters.
    \item That gives us \textbf{aggregates}: goods demand, capital demand, labour supply, \textbf{as a function of} prices, parameters, market tightness, etc.
    \item In a sense, you can think of the household block as representing \textbf{one side} of the economy, sort of like a big aggregate demand curve.
  \end{itemize}
\end{frame}

\begin{frame}{Why Stages?}
  \begin{itemize}
    \item Stages are \textbf{composable}: complex models built by combining Stages.
    \item Stages have \textbf{precise mathematical definitions}: a model can be defined and solved without worrying about implementation details.
    \item Stages are \textbf{standardized}: optimized implementations exist that can be downloaded and reused (e.g.\ \texttt{HouseholdStages.jl}, the SSJ Toolkit, or ECON-ARK).
  \end{itemize}

\end{frame}

\begin{frame}{How the Household Block Fits In}
  The \textbf{household block} (defined and solved using \textbf{Stages}) is one part of a model.

  You can think of the rest of a model as consisting of:
  \begin{itemize}
    \item A system of equations describing the rest of the economy---firms, government, etc.---often implemented as a \textbf{directed acyclic graph} (DAG) of functions.
    \item A series of \textbf{Outer Loops} that solve for equilibrium prices, steady states, transition paths, or global solutions with aggregate uncertainty.
  \end{itemize}  
\end{frame}

\begin{frame}{Outer Loops: Examples}
  Equilibrium: Search over prices against \textbf{market clearing} or \textbf{excess demand} objective.

  Steady State: Search over population distributions $\Lambda$ against \textbf{population-evolution} $\Lambda^\text{end} - \Lambda^{\text{start}'}$ objective.

  Calibration: Search over parameters against \textbf{model fit} $\text{moment}^\text{model} - \text{moment}^\text{data}$ objective.

  Perfect Foresight Transition: Search over \textbf{paths} of market-clearing prices against path-of-excess-demand objective.

  Aggregate Uncertainty: Search over \textbf{global value functions} against \textbf{outer Bellman objective}.
\end{frame}

\begin{frame}{Aiyagari Model}
  The Aiyagari model has these three blocks:
  \begin{enumerate}
    \item A \textbf{household block} made of \textbf{Stages}.
    \item A \textbf{firm} that turns capital and labour into goods.
    \item A notion of \textbf{equilibrium} (steady state).
  \end{enumerate}

  Familiar economics:
  \begin{enumerate}
    \item A demand side.
    \item A supply side.
    \item A notion of equilibrium.
  \end{enumerate}
\end{frame}

\section{Aiyagari Household Block}
% =============================================================

\begin{frame}{Same As Last Time}

  The three-Stage household block from last time:
  \begin{enumerate}
    \item Income shock (Markov on $z$).
    \item Income (cash-on-hand $b = R\,b^{\mathrm{end}} + z$).
    \item Consumption--saving (grid search on $b^{\mathrm{end}}$).
  \end{enumerate}

  Steady-state solution given prices:
  \begin{itemize}
    \item VFI: Value function $V$ and policy $c^\star$.
    \item Forward simulation: Stationary distribution $\lambda^\star$.
    \item Today: Make $R$ an endogenous \emph{price} we have to solve for.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Solve the household block at a given $R$}
  \begin{lstlisting}
  function solve_aiyagari_steady_state(params; verbosity=1)
      V_end = solve_vfi(params; verbosity)
      return (;V_end, λ=find_steady_state_population(V_end, params; verbosity))
  end
  \end{lstlisting}

  \begin{itemize}
    \item \texttt{params} contains prices and parameters.
    \item In one function call, solve for steady state.
    \item Can think of as, among other things, one big function $R \mapsto \bar C - Y$ (excess goods demand at given prices).
  \end{itemize}
\end{frame}

\section{Aggregates}
% =============================================================

\begin{frame}{Aggregate Moments as Inner Products}
  Every aggregate moment is a \textbf{dual vector}: an inner product with $\lambda^\star$.
  \begin{align*}
    \bar K &= \sum_{i,j} b_i \, \lambda^\star_{ij} \\
    \bar L &= \sum_{i,j} z_j \, \lambda^\star_{ij}        \\
    \bar C &= \sum_{i,j} c^\star_{ij} \, \lambda^\star_{ij} \\
    \bar V &= \sum_{i,j} V_{ij} \, \lambda^\star_{ij}
  \end{align*}
\end{frame}

\begin{frame}[fragile]{\texttt{compute\_household\_aggregates}}
  \begin{lstlisting}
  function compute_household_aggregates(;param_vals...)
      params = get_params(;param_vals...)
      (;V_end, λ) = solve_aiyagari_steady_state(params)

      b_inds_new = policy_function(V_end, params)
      c_policy = params.b_grid .- params.b_grid[b_inds_new]

      L_agg = sum(params.z_grid' .* λ)
      K_agg = sum(params.b_grid .* λ)
      V_agg = sum(V_end .* λ)
      C_agg = sum(c_policy .* λ)

      return (;V_end, λ, L_agg, K_agg, V_agg, C_agg)
  end
  \end{lstlisting}
\end{frame}

\section{The Firm}
% =============================================================

\begin{frame}[fragile]{Cobb--Douglas}
  Representative firm with Cobb--Douglas production:
  \[
    Y = K^\alpha L^{1-\alpha}.
  \]
  This is the supply side of the model.

  In code,
  \begin{lstlisting}
  compute_Y(K, L; α=1/3) = K^α * L^(1-α)
  \end{lstlisting}
\end{frame}

\section{Equilibrium}
% =============================================================

\begin{frame}{Goods market clearing}
  In steady state,
  \[
    \bar C \;=\; Y.
  \]
  Define the \emph{excess demand} for goods:
  \[
    \text{excess\_demand}(R) = \frac{\bar C(R) - Y(R)}{Y(R)}.
  \]
  Equilibrium $R$ is the \textbf{root} of this function: The value solving $\text{excess\_demand}(R) = 0$.
\end{frame}

\begin{frame}[fragile]{\texttt{compute\_excess\_demand}}
  We can code up this function,
  \begin{lstlisting}
  function compute_excess_demand(R)
      (;K_agg, L_agg, C_agg) = compute_household_aggregates(;R)
      Y_agg = compute_Y(K_agg, L_agg)
      return excess_demand = (C_agg - Y_agg)/Y_agg
  end
  \end{lstlisting}
  The chain of direct effects is:
  \[
    R \to V \to c^\star \to \lambda^\star \to (\bar K, \bar C) \to Y.
  \]
\end{frame}

\begin{frame}[fragile]{Solving for equilibrium $R$}
  \begin{lstlisting}
  tol = 0.01
  update_speed = 0.001

  R = 1.01
  err = Inf
  while err > tol
      excess_demand = compute_excess_demand(R)

      R_new = R + excess_demand*update_speed
      err = abs(excess_demand)

      R = R_new

      println("R=$R\t excess_demand=$excess_demand")
  end
  \end{lstlisting}

  \begin{itemize}
    \item Tatonnement: A simple gradient-style update: update $R$ in the direction of excess goods demand.
    \item If $\bar C > Y$, households consuming too much --- $\uparrow R\Rightarrow \uparrow$ saving $\Rightarrow \uparrow K, Y$.
    \item If $\bar C < Y$, households are saving too much --- $\downarrow R\Rightarrow \downarrow$ saving $\Rightarrow \downarrow K, Y$.
  \end{itemize}
\end{frame}
