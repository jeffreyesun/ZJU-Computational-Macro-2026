\section{Krusell-Smith 方法}
% =============================================================

\begin{frame}{设定：带总量 TFP 冲击的 Aiyagari 模型}
  总量 TFP $Z_t \in \{Z_{\text{bad}}, Z_{\text{good}}\}$ 服从转移矩阵为 $\Pi_Z$ 的马尔可夫链。

  特异性状态：$(z_{it}, b_{it})$
  \begin{itemize}
    \item 生产率 $z_{it} \in \{z_{\text{lo}}, z_{\text{hi}}\}$，服从转移矩阵为 $\Pi_z$ 的马尔可夫链
    \item 财富 $b_{it}$。
  \end{itemize}

  技术：带 TFP $Z_t$ 的 Cobb--Douglas 生产函数：
  \[
    Y_t = Z_t\, K_t^{\alpha} L^{1-\alpha}.
  \]
  假设 $L$ 恒定，等于 $z$ 的平稳均值。
\end{frame}

\begin{frame}{价格}
  毛利率与工资由 $Z_t, K_t, L$ 给定，
  \[
    R(Z_t, K_t) = 1 + \alpha\, Z_t\, (K_t/L)^{\alpha-1} - \delta,
  \]
  \[
    w(Z_t, K_t) = (1-\alpha)\, Z_t\, (K_t/L)^{\alpha}.
  \]

  问题：家庭的价格预测既\textbf{影响}未来的 $K_{t+1}$，又\textbf{依赖于}它。
\end{frame}

\begin{frame}{$K_{t+1}$ 依赖于 $\Lambda_t$}
  \[
    K_{t+1} = \int b_i'(b_i, z_i;\, Z_t, K_t)\, \mathrm d\Lambda_t(b_i, z_i).
  \]

  策略函数 $b_i'(b_i, z_i;\, Z_t, K_t)$ 依赖于价格预测。

  价格 $R_{t+s}, w_{t+s}$ 的预测通过 $K_{t+s}$ 依赖于 $\Lambda_{t+s}$。

  \textbf{注意：}来自不同 $\Lambda_t$ 的相同 $K_t$ 可能导出不同的 $K_{t+1}$。
\end{frame}

\begin{frame}{贝尔曼方程}
  \textbf{分布} $\Lambda$ 成为\textbf{状态变量}：
  \[
    V(b_i, z_i; \Lambda, Z) = \max_{c_i, b_i'} u(c_i) + \beta\, \E\!\bigl[V(b_i', z_i'; \Lambda', Z') \mid z_i, Z\bigr],
  \]
\end{frame}

\begin{frame}{$\Lambda$ 是无限维的}
  $\Lambda_t$ 是定义在所有 $(b, z)$ 上的\textbf{分布}。

  ``真实''的 $\Lambda_t$ 是无限维的。

  即便我们将 $\Lambda_t$ 离散化为一个矩阵，\textbf{每个网格点都是一个状态变量}。

  当财富网格点数 $N_b = 80$、生产率水平数 $\#z = 2$ 时，网格共有 160 个点。

  在所有可能的 $\Lambda$ 值上构建 $V$ 网格：$5^{160-1}$ 个单元，约为 $10^{111}$。
\end{frame}

\begin{frame}{对 $V(\ldots\ ;\ \ldots,\ \Lambda)$ 进行可处理的参数化}
  两种思路：

  \textbf{(a) 降低 $\Lambda$ 的维度。}（Krusell-Smith。）用 $\Lambda_t$ 的矩来代替整个分布。

  \textbf{(b) 降低表示的复杂度。}（nVFI。）用一族灵活的函数（例如\textbf{神经网络}）对 $V(\Lambda, \cdot)$ 进行参数化。
\end{frame}

% =============================================================
\section[K--S 方法]{Krusell--Smith 方法}
% =============================================================

\begin{frame}{微观基础：有限理性}
  Krusell and Smith (1998)：假设\textbf{家庭}只观测到 $\Lambda_t$ 的矩，而非整个分布。

  即，求解 $V(b, z; Z, K)$ 的模型，而非 $V(b, z; Z, \Lambda)$ 的模型。
\end{frame}

\begin{frame}{近似运动方程（ALM）}
  假设家庭对 $K$ 使用如下预测规则：
  \[
    \log K_{t+1} = a_0(Z_t) + a_1(Z_t) \log K_t.
  \]

  \textbf{每个 $Z$ 状态有两个系数。}对于 2 状态的 $Z$，共四个数。

  \textbf{对数线性。} K--S 发现这实际上拟合得非常好。
\end{frame}

\begin{frame}{有限理性的贝尔曼方程}
  在状态中用 $K$ 替换 $\Lambda$：
  \[
    V(b_i, z_i, Z, K) = \max_{c_i, b_i'} u(c_i) + \beta\, \E\bigl[V(b_i', z_i', Z', K')\bigr],
  \]
  约束条件为
  \[
    c_i + b_i' = R(Z, K)\, b_i + w(Z, K)\, z_i,
  \]
  主体的\emph{信念}为
  \[
    K' = \exp(a_0(Z) + a_1(Z) \log K),
  \]
  且 $Z'$ 从 $\Pi_Z$ 中抽取。
\end{frame}

\begin{frame}{行为链（五个阶段）}
  在近似运动方程（ALM）的假设下，我们只需向家庭模块添加两个阶段即可求解模型：
  \begin{enumerate}
    \item \texttt{z\_shock}：$z_{it} \to z_i^{t+1}$（在 \texttt{:z} 上的马尔可夫转移）。
    \item \texttt{income}：$b_i' \gets R\, b_i + w\, z_i$，其中 $(R, w)$ 由 $(Z, K)$ 给出。
    \item \texttt{savings}：选择 $b_i^{t+1}$。
    \item \texttt{K\_evolve}：$K_t \to K_{t+1}$。
    \item \texttt{Z\_shock}：$Z_t \to Z_{t+1}$。
  \end{enumerate}

  \begin{center}\small
  \texttt{z\_shock} $\circ$ \texttt{income} $\circ$ \texttt{savings} $\circ$ \texttt{K\_evolve} $\circ$ \texttt{Z\_shock}.
  \end{center}

  求解这条链便给出了在给定 $a_0(\cdot), a_1(\cdot)$ 下家庭的\textbf{行为}，并使我们能够向前模拟\textbf{真实}模型。
\end{frame}

\begin{frame}{K--S 外层循环}
  \begin{enumerate}
    \item \textbf{选取}一个初始 ALM $(a_0^{(0)}, a_1^{(0)})$。
    \item 在当前 ALM 下\textbf{求解}四维家庭 VFI。
    \item 在实现的 $\{Z_t\}$ 与\emph{真实}加总 $K_t = \int b\, \mathrm d\Lambda_t$ 下\textbf{模拟}一个长时段经济。
    \item 通过将 $\log K_{t+1}$ 对 $\log K_t$ 做 OLS（按 $Z$ 分别进行）来\textbf{重新拟合} $(a_0, a_1)$。
    \item \textbf{迭代}直至 ALM 系数不再变动。
  \end{enumerate}
\end{frame}

\section{实现 Krusell--Smith}
% =============================================================

\begin{frame}{参数}
  \textbf{Krusell--Smith 校准}（1998）：
  \begin{itemize}
    \item $\beta = 0.96$，$\gamma = 1$（对数效用），$\alpha = 0.36$，$\delta = 0.025$。
    \item $Z \in \{0.99, 1.01\}$，$\Pi_Z$ 具有持续性（对角线上为 $0.875$）。
    \item $z \in \{0.07, 1.0\}$。
  \end{itemize}

  \textbf{网格}（为示例起见取较小规模）：
  \begin{itemize}
    \item $N_b = 80$，对数等距分布于 $[0, 80]$。
    \item $N_K = 5$，线性分布于 $[10.5, 14.5]$。
    \item 单元数：$80 \times 2 \times 2 \times 5 = 1600$。
  \end{itemize}
\end{frame}

  \begin{lstlisting}
  layout = StateLayout(

\end{frame}

\begin{frame}[fragile]{阶段 1--2：z\_shock 与 income \\ 阶段 1--2}
  \begin{lstlisting}
  z_shock = MarkovStage(layout; axis=:z, transition=p.Π_z)

  income = WealthChangeStage(layout; wealth_post=(cell; env) -> begin

  end)
  \end{lstlisting}

  \textbf{注意。}这从\emph{单元坐标}中读取 $(Z, K)$。
\end{frame}

  \begin{itemize}
    \item 分治搜索利用了关于财富的单调性。
  \end{itemize}
\end{frame}

  \begin{lstlisting}

      wealth_post=(cell; env) ->
          exp(env.a0[cell.Z_idx] + env.a1[cell.Z_idx] * log(cell.K)))

  Z_shock = MarkovStage(layout; axis=:Z_idx, transition=p.Π_Z)
  \end{lstlisting}
\end{frame}

\begin{frame}[fragile]{构建家庭模块}
  \begin{lstlisting}
  chain = z_shock ∘ income ∘ savings ∘ K_evolve ∘ Z_shock

  hh = define_moments!(chain;
      K_supplied=at_end(integrand=:wealth, reduce=sum))
  \end{lstlisting}
\end{frame}

\begin{frame}[fragile]{内层求解：给定 $a_0(\cdot), a_1(\cdot)$ 的 VFI}
  \begin{lstlisting}
  a0_init = [log(K_bar), log(K_bar)]

  \textbf{初始猜测。}一个\emph{常数} ALM：对两个 $Z$ 状态均有 $K_{t+1} = \bar K$（$a_1 \equiv 0$，$a_0 \equiv \log \bar K$）。
\end{frame}

\begin{frame}{模拟用家庭模块}

  在模拟时，我们想要的是\emph{实现}的 $(Z_t, K_t)$，而非主体的信念。

  行为链中的 K\_evolve 与 Z\_shock 编码的是\emph{信念}，而非 $K$ 的真实运动方程。

  真实的链只包含前三个阶段（\texttt{z\_shock} $\circ$ \texttt{income} $\circ$ \texttt{savings}），而 $Z, K$ 在\textbf{外层循环}中更新。
\end{frame}

\begin{frame}{模拟真实模型}
  每一期：
  \begin{enumerate}
    \item 计算 $K_t = \sum_{b, z} b\, \Lambda_t[b, z]$。
    \item 将 $\Lambda_t$ 提升为四维分布，把 $(Z_t, K_t)$ 添加为``特异性''状态变量。
    \item 使用行为链（5 个阶段）模拟该四维分布。
    \item 对 $(Z, K)$ 求边际，得到 $(b, z)$ 上的 $\Lambda_{t+1}$。
    \item 从 $\Pi_Z$ 中抽取 $Z_{t+1} \mid Z_t$。
  \end{enumerate}
\end{frame}

  function ks_iterate(p::KSParams, K_bar, Λ_0; update_speed=0.5, rtol=1e-3, maxiter=20, T_sim=2000, T_burn=500)
      hh_4d = behavioural_alm_household(p)
      a0 = [log(K_bar), log(K_bar)]
      a1 = [0.0, 0.0]
      err, iters = Inf, 0
      while err >= rtol
          env = make_env(hh_4d; a0, a1)
          solve_steady_state_given_env!(hh_4d, env; lambda_tol=1e-5, lambda_maxiter=50_000)

          sim = ks_simulate(hh_4d, p; T=T_sim, Λ_init=Λ_0, Z_init=1)
          window_fit = (T_burn + 1):(T_sim - 1)
          fit_bad  = regress_ALM(sim.K_path, sim.Z_idx_path, 1, window_fit)
          fit_good = regress_ALM(sim.K_path, sim.Z_idx_path, 2, window_fit)

          a0_new = [fit_bad.a0, fit_good.a0]
          a1_new = [fit_bad.a1, fit_good.a1]
          err = maximum(abs.(vcat(a0_new .- a0, a1_new .- a1)))

          a0 = (1 - update_speed) .* a0 .+ update_speed .* a0_new
          a1 = (1 - update_speed) .* a1 .+ update_speed .* a1_new
          iters += 1
      end
      return (; a0, a1, sim, fit_bad, fit_good, iters)
  end
  \end{lstlisting}
